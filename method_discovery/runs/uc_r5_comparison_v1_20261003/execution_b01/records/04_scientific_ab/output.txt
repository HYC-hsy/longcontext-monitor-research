package streaming

import (
\t\"context\"

\t\"github.com/cloudwego/kitex/pkg/stats\"
)

// Header represents RPC header metadata
type Header map[string]string

// Trailer represents RPC trailer metadata
type Trailer map[string]string

// ClientStream represents the client side of a streaming RPC
type ClientStream interface {
\t// SendMsg sends a message to the peer
\tSendMsg(ctx context.Context, m any) error
\t// RecvMsg receives a message from the peer
\tRecvMsg(ctx context.Context, m any) error
\t// Header returns header metadata from the server
\tHeader() (Header, error)
\t// Trailer returns trailer metadata from the server
\tTrailer() (Trailer, error)
\t// CloseSend closes the send direction
\tCloseSend(ctx context.Context) error
\t// Context returns the stream's context
\tContext() context.Context
}

// ServerStream represents the server side of a streaming RPC
type ServerStream interface {
\t// SendMsg sends a message to the peer
\tSendMsg(ctx context.Context, m any) error
\t// RecvMsg receives a message from the peer
\tRecvMsg(ctx context.Context, m any) error
\t// SetHeader sets header metadata (may be called multiple times; merged)
\tSetHeader(hd Header) error
\t// SendHeader sends header metadata (fails if called after first response)
\tSendHeader(hd Header) error
\t// SetTrailer sets trailer metadata (merged when called multiple times)
\tSetTrailer(hd Trailer) error
}

// CloseCallbackRegister allows registering a callback to be called when the stream closes
type CloseCallbackRegister interface {
\tRegisterCloseCallback(cb func(error))
}

// GRPCStreamGetter provides access to the underlying gRPC stream
type GRPCStreamGetter interface {
\tGetGRPCStream() Stream
}

// EventHandler is a function that handles stream events
type EventHandler func(ctx context.Context, evt stats.Event, err error)
"
}
