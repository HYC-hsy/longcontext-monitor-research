// Header represents RPC metadata headers.
type Header map[string]string

// Trailer represents RPC metadata trailers.
type Trailer map[string]string

// ClientStream represents a client-side stream with methods for sending and receiving messages.
type ClientStream interface {
\t// SendMsg sends a message to the peer.
\tSendMsg(ctx context.Context, m any) error
\t// RecvMsg receives a message from the peer.
\tRecvMsg(ctx context.Context, m any) error
\t// Header returns header metadata from the server.
\tHeader() (Header, error)
\t// Trailer returns trailer metadata from the server.
\tTrailer() (Trailer, error)
\t// CloseSend closes the send direction.
\tCloseSend(ctx context.Context) error
\t// Context returns the stream's context.
\tContext() context.Context
}

// ServerStream represents a server-side stream with methods for sending and receiving messages.
type ServerStream interface {
\t// SendMsg sends a message to the peer.
\tSendMsg(ctx context.Context, m any) error
\t// RecvMsg receives a message from the peer.
\tRecvMsg(ctx context.Context, m any) error
\t// SetHeader sets header metadata (may be called multiple times; merged).
\tSetHeader(hd Header) error
\t// SendHeader sends header metadata (fails if called after first response).
\tSendHeader(hd Header) error
\t// SetTrailer sets trailer metadata (merged when called multiple times).
\tSetTrailer(hd Trailer) error
}

// CloseCallbackRegister allows registering a callback to be invoked when the stream closes.
type CloseCallbackRegister interface {
\t// RegisterCloseCallback registers a callback function that will be called when the stream closes.
\tRegisterCloseCallback(cb func(error))
}

// GRPCStreamGetter provides access to the underlying gRPC stream.
type GRPCStreamGetter interface {
\t// GetGRPCStream returns the underlying Stream interface.
\tGetGRPCStream() Stream
}

// EventHandler is a function type for handling stream events.
type EventHandler func(ctx context.Context, evt stats.Event, err error)
"
}

## Architecture Notes

### Streaming Options Architecture
- `client.Options` (internal/client) - global client configuration
- `client.StreamOptions` (internal/client) - streaming-specific settings
- `callopt.CallOptions` (client/callopt) - per-call options wrapper
- `streamcall.CallOption` - streaming call option pattern

The bridge: `callopt.CallOptions.StreamOptions` field connects the streaming call option system to internal client streaming configuration.

### Retry Options Migration Context
- **UnaryOptions** contains unary-specific retry/fallback configuration
- Monitor required migrating retry configuration from top-level `Options` fields to nested `UnaryOptions` fields
- This architectural change improves separation between unary and streaming retry logic
