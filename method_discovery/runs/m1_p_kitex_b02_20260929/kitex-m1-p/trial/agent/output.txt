          conns = append(conns, map[string]interface{}{
              \"address\":      c.RemoteAddr().String(),
              \"state\":        c.GetState().String(),
              \"active_streams\": c.ActiveStreams(),
              \"is_closing\":   c.IsClosing(),
          })
      }
      
      return map[string]interface{}{
          \"size\":         p.size,
          \"connections\":  conns,
          \"total_conns\":  len(p.conns),
      }
  }
