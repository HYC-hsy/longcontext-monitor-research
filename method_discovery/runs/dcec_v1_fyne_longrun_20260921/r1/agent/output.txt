// parseHexColor parses a hex color string in various formats:
// 3 digits: \"abc\" -> 0xaa 0xbb 0xcc 0xff
// 4 digits: \"abcd\" -> 0xaa 0xbb 0xcc 0xdd
// 6 digits: \"a1b2c3\" -> 0xa1 0xb2 0xc3 0xff
// 8 digits: \"a1b2c3f4\" -> 0xa1 0xb2 0xc3 0xf4
// All formats support optional \"#\" prefix
func parseHexColor(s string) (color.Color, error) {
\ts = strings.TrimSpace(s)
\ts = strings.TrimPrefix(s, \"#\")

\tvar r, g, b, a uint8
\ta = 255 // default alpha

\tswitch len(s) {
\tcase 3:
\t\t// RGB format: expand each digit
\t\tvals, err := parseHexDigits(s)
\t\tif err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\t\tr = vals[0]*16 + vals[0]
\t\tg = vals[1]*16 + vals[1]
\t\tb = vals[2]*16 + vals[2]

\tcase 4:
\t\t// RGBA format: expand each digit
\t\tvals, err := parseHexDigits(s)
\t\tif err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\t\tr = vals[0]*16 + vals[0]
\t\tg = vals[1]*16 + vals[1]
\t\tb = vals[2]*16 + vals[2]
\t\ta = vals[3]*16 + vals[3]

\tcase 6:
\t\t// RRGGBB format
\t\tval, err := strconv.ParseUint(s, 16, 32)
\t\tif err != nil {
\t\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color: %s\", s)
\t\t}
\t\tr = uint8((val >> 16) & 0xFF)
\t\tg = uint8((val >> 8) & 0xFF)
\t\tb = uint8(val & 0xFF)

\tcase 8:
\t\t// RRGGBBAA format
\t\tval, err := strconv.ParseUint(s, 16, 64)
\t\tif err != nil {
\t\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color: %s\", s)
\t\t}
\t\tr = uint8((val >> 24) & 0xFF)
\t\tg = uint8((val >> 16) & 0xFF)
\t\tb = uint8((val >> 8) & 0xFF)
\t\ta = uint8(val & 0xFF)

\tdefault:
\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color length: %s\", s)
\t}

\treturn color.NRGBA{R: r, G: g, B: b, A: a}, nil
}

func parseHexDigits(s string) ([]uint8, error) {
\tvals := make([]uint8, len(s))
\tfor i, c := range s {
\t\tvar val uint8
\t\tswitch {
\t\tcase c >= '0' && c <= '9':
\t\t\tval = uint8(c - '0')
\t\tcase c >= 'a' && c <= 'f':
\t\t\tval = uint8(c - 'a' + 10)
\t\tcase c >= 'A' && c <= 'F':
\t\t\tval = uint8(c - 'A' + 10)
\t\tdefault:
\t\t\treturn nil, fmt.Errorf(\"invalid hex digit: %c\", c)
\t\t}
\t\tvals[i] = val
\t}
\treturn vals, nil
}
