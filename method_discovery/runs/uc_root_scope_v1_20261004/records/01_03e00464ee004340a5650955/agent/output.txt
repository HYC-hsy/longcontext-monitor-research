import (
\t\"fmt\"
\t\"sync\"

// sprintfBinding is a String binding that formats values from multiple source bindings
type sprintfBinding struct {
\tbase
\tformat  string
\tsources []DataItem
\tlock    sync.RWMutex
}

// NewSprintf creates a new String binding that formats its content using fmt.Sprintf with the
// given format string and source bindings. The binding listens to all sources and updates
// automatically when any source changes.
//
// Example:
//   name := binding.NewString()
//   age := binding.NewInt()
//   formatted := binding.NewSprintf(\"Name: %s, Age: %d\", name, age)
//
// Since: 2.5
func NewSprintf(format string, sources ...DataItem) String {
\tb := &sprintfBinding{
\t\tformat:  format,
\t\tsources: sources,
\t}
\t
