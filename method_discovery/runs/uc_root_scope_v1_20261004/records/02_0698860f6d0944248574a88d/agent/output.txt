// FromJSON parses a JSON string and returns a Theme.

// Since: 2.2
func FromJSON(data string) (fyne.Theme, error) {
\treturn FromJSONReader(strings.NewReader(data))

// FromJSONReader parses JSON from an io.Reader and returns a Theme.
// On parse error, returns the default theme along with the error.
//
// Since: 2.2
func FromJSONReader(r io.Reader) (fyne.Theme, error) {
\tvar spec jsonThemeSpec
\tdecoder := json.NewDecoder(r)
\tif err := decoder.Decode(&spec); err != nil {
\t\treturn Default(), err
\t}

\ttheme := &jsonTheme{
\t\tcolors:      make(map[string]color.Color),
\t\tcolorsDark:  make(map[string]color.Color),
\t\tcolorsLight: make(map[string]color.Color),
\t\tsizes:       make(map[string]float32),
\t\tfonts:       make(map[string]fyne.Resource),
\t\ticons:       make(map[string]fyne.Resource),
\t}

\t// Parse generic colors
\tfor name, hexStr := range spec.Colors {
\t\tcol, err := parseHexColor(hexStr)
\t\tif err != nil {
\t\t\tfyne.LogError(fmt.Sprintf(\"Invalid color hex for %s: %s\", name, hexStr), err)
\t\t\tcontinue
\t\t}
\t\ttheme.colors[name] = col
\t}

\t// Parse dark variant colors
\tfor name, hexStr := range spec.ColorsDark {
\t\tcol, err := parseHexColor(hexStr)
\t\tif err != nil {
\t\t\tfyne.LogError(fmt.Sprintf(\"Invalid color hex for dark.%s: %s\", name, hexStr), err)
\t\t\tcontinue
\t\t}
\t\ttheme.colorsDark[name] = col
\t}

\t// Parse light variant colors
\tfor name, hexStr := range spec.ColorsLight {
\t\tcol, err := parseHexColor(hexStr)
\t\tif err != nil {
\t\t\tfyne.LogError(fmt.Sprintf(\"Invalid color hex for light.%s: %s\", name, hexStr), err)
\t\t\tcontinue
\t\t}
\t\ttheme.colorsLight[name] = col
\t}

\t// Parse sizes
\tfor name, value := range spec.Sizes {
\t\ttheme.sizes[name] = value
\t}

\t// Parse fonts
\tfor style, uri := range spec.Fonts {
\t\tres, err := loadResourceFromURI(uri)
\t\tif err != nil {
\t\t\tfyne.LogError(fmt.Sprintf(\"Failed to load font %s from %s\", style, uri), err)
\t\t\tcontinue
\t\t}
\t\ttheme.fonts[style] = res
\t}

\t// Parse icons
\tfor name, uri := range spec.Icons {
\t\tres, err := loadResourceFromURI(uri)
\t\tif err != nil {
\t\t\tfyne.LogError(fmt.Sprintf(\"Failed to load icon %s from %s\", name, uri), err)
\t\t\tcontinue
\t\t}
\t\ttheme.icons[name] = res
\t}

\treturn theme, nil
}

type jsonThemeSpec struct {
\tColors      map[string]string  `json:\"Colors\"`
\tColorsDark  map[string]string  `json:\"Colors-dark\"`
\tColorsLight map[string]string  `json:\"Colors-light\"`
\tSizes       map[string]float32 `json:\"Sizes\"`
\tFonts       map[string]string  `json:\"Fonts\"`
\tIcons       map[string]string  `json:\"Icons\"`
}

type jsonTheme struct {
\tcolors      map[string]color.Color
\tcolorsDark  map[string]color.Color
\tcolorsLight map[string]color.Color
\tsizes       map[string]float32
\tfonts       map[string]fyne.Resource
\ticons       map[string]fyne.Resource
}

func (t *jsonTheme) Color(name fyne.ThemeColorName, variant fyne.ThemeVariant) color.Color {
\tnameStr := string(name)

\t// Check variant-specific colors first
\tif variant == fyne.VariantDark {
\t\tif col, ok := t.colorsDark[nameStr]; ok {
\t\t\treturn col
\t\t}
\t} else if variant == fyne.VariantLight {
\t\tif col, ok := t.colorsLight[nameStr]; ok {
\t\t\treturn col
\t\t}
\t}

\t// Check generic colors
\tif col, ok := t.colors[nameStr]; ok {
\t\treturn col
\t}

\t// Fallback to default theme
\treturn Default().Color(name, variant)
}

func (t *jsonTheme) Font(style fyne.TextStyle) fyne.Resource {
\tvar key string
\tif style.Monospace {
\t\tkey = \"monospace\"
\t} else if style.Bold && style.Italic {
\t\tkey = \"boldItalic\"
\t} else if style.Bold {
\t\tkey = \"bold\"
\t} else if style.Italic {
\t\tkey = \"italic\"
\t} else {
\t\tkey = \"regular\"
\t}

\tif font, ok := t.fonts[key]; ok {
\t\treturn font
\t}

\t// Fallback to default theme
\treturn Default().Font(style)
}

func (t *jsonTheme) Icon(name fyne.ThemeIconName) fyne.Resource {
\tif icon, ok := t.icons[string(name)]; ok {
\t\treturn icon
\t}

\t// Fallback to default theme
\treturn Default().Icon(name)
