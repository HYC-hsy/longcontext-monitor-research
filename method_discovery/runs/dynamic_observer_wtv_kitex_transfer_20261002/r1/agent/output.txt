// Header represents streaming headers as a simple string map.
// Keys are case-insensitive and normalized to lowercase.
type Header map[string]string

// Trailer represents streaming trailers as a simple string map.
// Keys are case-insensitive and normalized to lowercase.
type Trailer map[string]string

// ClientStream represents the client-side streaming interface.
type ClientStream interface {
\t// Context returns the context for this stream.
\tContext() context.Context

\t// SendMsg sends a message on the stream.
\t// It blocks until the message is sent or an error occurs.
\tSendMsg(m interface{}) error

\t// RecvMsg receives a message from the stream.
\t// It blocks until a message is available or an error occurs.
\t// Returns io.EOF when the stream is closed by the server.
\tRecvMsg(m interface{}) error

\t// CloseSend closes the send direction of the stream.
\t// After calling CloseSend, no more messages can be sent.
\tCloseSend() error

\t// Header returns the header metadata from the server.
\t// It blocks if the header has not been received yet.
\tHeader() (Header, error)

\t// Trailer returns the trailer metadata from the server.
\t// It must only be called after RecvMsg returns io.EOF.
\tTrailer() Trailer
}

// ServerStream represents the server-side streaming interface.
type ServerStream interface {
\t// Context returns the context for this stream.
\tContext() context.Context

\t// SendMsg sends a message on the stream.
\t// It blocks until the message is sent or an error occurs.
\tSendMsg(m interface{}) error

\t// RecvMsg receives a message from the stream.
\t// It blocks until a message is available or an error occurs.
\t// Returns io.EOF when the stream is closed by the client.
\tRecvMsg(m interface{}) error

\t// SetHeader sets the header metadata to be sent to the client.
\t// It can be called multiple times before SendHeader or the first SendMsg.
\t// All provided metadata will be merged.
\tSetHeader(Header) error

\t// SendHeader sends the header metadata to the client.
\t// It can only be called once and must be called before the first SendMsg.
\tSendHeader(Header) error

\t// SetTrailer sets the trailer metadata to be sent to the client.
\t// It can be called multiple times.
\t// The trailer will be sent when the stream is closed.
\tSetTrailer(Trailer)
}
