// parseHexColor parses a hex color string into a color.Color.
// Supports formats: #rgb, #rgba, #rrggbb, #rrggbbaa (with or without # prefix)
func parseHexColor(hex string) (color.Color, error) {
\t// Remove # prefix if present
\thex = strings.TrimPrefix(hex, \"#\")

\tvar r, g, b, a uint8
\ta = 0xff // default alpha

\tswitch len(hex) {
\tcase 3: // rgb
\t\tvals, err := parseHexDigits(hex, 1)
\t\tif err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\t\tr = vals[0] | (vals[0] << 4)
\t\tg = vals[1] | (vals[1] << 4)
\t\tb = vals[2] | (vals[2] << 4)

\tcase 4: // rgba
\t\tvals, err := parseHexDigits(hex, 1)
\t\tif err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\t\tr = vals[0] | (vals[0] << 4)
\t\tg = vals[1] | (vals[1] << 4)
\t\tb = vals[2] | (vals[2] << 4)
\t\ta = vals[3] | (vals[3] << 4)

\tcase 6: // rrggbb
\t\tvals, err := parseHexDigits(hex, 2)
\t\tif err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\t\tr = vals[0]
\t\tg = vals[1]
\t\tb = vals[2]

\tcase 8: // rrggbbaa
\t\tvals, err := parseHexDigits(hex, 2)
\t\tif err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\t\tr = vals[0]
\t\tg = vals[1]
\t\tb = vals[2]
\t\ta = vals[3]

\tdefault:
\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color format: %s\", hex)
\t}

\treturn color.NRGBA{R: r, G: g, B: b, A: a}, nil
}

// parseHexDigits parses hex string into byte values
func parseHexDigits(hex string, digitsPer int) ([]uint8, error) {
\tcount := len(hex) / digitsPer
\tresult := make([]uint8, count)

\tfor i := 0; i < count; i++ {
\t\tstart := i * digitsPer
\t\tend := start + digitsPer
\t\tval, err := strconv.ParseUint(hex[start:end], 16, 8)
\t\tif err != nil {
\t\t\treturn nil, err
\t\t}
\t\tresult[i] = uint8(val)
\t}

\treturn result, nil
}
