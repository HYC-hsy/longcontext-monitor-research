package streaming

import (
\t\"context\"
)

// Header represents streaming metadata headers.
type Header map[string]string

// Trailer represents streaming metadata trailers.
type Trailer map[string]string

// ClientStream defines the typed client-side streaming interface.
type ClientStream[Req, Resp any] interface {
\t// Context returns the stream context.
\tContext() context.Context

\t// Send sends a request message to the server.
\tSend(req *Req) error

\t// Recv receives a response message from the server.
\tRecv(resp *Resp) error
