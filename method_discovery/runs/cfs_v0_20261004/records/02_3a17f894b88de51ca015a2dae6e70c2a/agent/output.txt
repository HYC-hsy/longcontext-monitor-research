\t// Remove # prefix if present
\thex = strings.TrimPrefix(hex, \"#\")

\tvar r, g, b, a uint8

\tswitch len(hex) {
\tcase 3:
\t\t// RGB: each digit doubled (abc -> aabbcc)
\t\tvals := make([]uint8, 3)
\t\tfor i := 0; i < 3; i++ {
\t\t\tval, err := strconv.ParseUint(hex[i:i+1], 16, 8)
\t\t\tif err != nil {
\t\t\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color: %s\", hex)
\t\t\t}
\t\t\tvals[i] = uint8(val*17) // 0xF -> 0xFF
\t\t}

\t\t\t}

// sprintf is a String binding that formats values from other bindings using fmt.Sprintf
type sprintf struct {
\tbase
\tformat string
\tvalues []DataItem
}

// NewSprintf creates a new String binding that formats the values from the provided bindings
// using fmt.Sprintf with the given format string. When any of the source bindings change,
// this binding updates with the newly formatted string.
//
// Example:
//   name := binding.NewString()
//   name.Set(\"Alice\")
//   age := binding.NewInt()
//   age.Set(30)
//   formatted := binding.NewSprintf(\"Name: %s, Age: %d\", name, age)
//   // formatted.Get() returns \"Name: Alice, Age: 30\"
//
// Since: 2.2
func NewSprintf(format string, bindings ...DataItem) String {
\ts := &sprintf{
\t\tformat: format,
\t\tvalues: bindings,
\t}

\t// Listen to all source bindings for changes
