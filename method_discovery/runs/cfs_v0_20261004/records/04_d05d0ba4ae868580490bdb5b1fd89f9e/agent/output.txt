````text
func (fr *Framer) readAndCheckFrameHeader() (http2.FrameHeader, error) {
    buf, err := fr.reader.Next(frameHeaderLen)
    if err != nil {
        return http2.FrameHeader{}, err
    }
    
    // Parse frame header
    length := uint32(buf[0])<<16 | uint32(buf[1])<<8 | uint32(buf[2])
    frameType := http2.FrameType(buf[3])
    flags := http2.Flags(buf[4])
    streamID := binary.BigEndian.Uint32(buf[5:]) & (1<<31 - 1)
    
    fh := http2.FrameHeader{
        Length:   length,
        Type:     frameType,
        Flags:    flags,
        StreamID: streamID,
    }
    
    // Validate: Invalid frame type (0x0A-0xEF reserved range)
    if frameType > 0x9 && frameType < 0xf0 {
        return fh, fr.connError(http2.ErrCodeProtocol,
            fmt.Sprintf(\"invalid frame type 0x%x (header hex: %02x %02x %02x %02x %02x %02x %02x %02x %02x)\",
                frameType, buf[0], buf[1], buf[2], buf[3], buf[4], buf[5], buf[6], buf[7], buf[8]))
    }
    
    // Validate: Reserved bit must be 0
    if buf[5]&0x80 != 0 {
        return fh, fr.connError(http2.ErrCodeProtocol,
            fmt.Sprintf(\"reserved bit set in stream ID (header hex: %02x %02x %02x %02x %02x %02x %02x %02x %02x)\",
                buf[0], buf[1], buf[2], buf[3], buf[4], buf[5], buf[6], buf[7], buf[8]))
    }
    
    // Validate: Frame length limit
    if length > fr.maxReadSize {
        return fh, fr.connError(http2.ErrCodeFrameSize,
            fmt.Sprintf(\"frame too large: %d > max %d (header hex: %02x %02x %02x %02x %02x %02x %02x %02x %02x)\",
                length, fr.maxReadSize, buf[0], buf[1], buf[2], buf[3], buf[4], buf[5], buf[6], buf[7], buf[8]))
    }
    
    return fh, nil
}
```
````
