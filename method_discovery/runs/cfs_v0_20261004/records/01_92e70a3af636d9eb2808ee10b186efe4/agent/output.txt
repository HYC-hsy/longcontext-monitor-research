// parseHexColor parses a hex color string into a color.Color
// Supports 3, 4, 6, and 8 digit hex formats, with or without # prefix
func parseHexColor(hex string) (color.Color, error) {
\t// Remove # prefix if present
\thex = strings.TrimPrefix(hex, \"#\")

\t// Validate hex characters
\tfor _, c := range hex {
\t\tif !((c >= '0' && c <= '9') || (c >= 'a' && c <= 'f') || (c >= 'A' && c <= 'F')) {
\t\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color: %s\", hex)
\t\t}
\t}

\tvar r, g, b, a uint8
\ta = 0xff // default alpha

\tswitch len(hex) {
\tcase 3:
\t\t// RGB format: \"abc\" -> 0xaa 0xbb 0xcc 0xff
\t\tr = hexDigitToByte(hex[0])
\t\tr = r<<4 | r
\t\tg = hexDigitToByte(hex[1])
\t\tg = g<<4 | g
\t\tb = hexDigitToByte(hex[2])
\t\tb = b<<4 | b

\tcase 4:
\t\t// RGBA format: \"abcd\" -> 0xaa 0xbb 0xcc 0xdd
\t\tr = hexDigitToByte(hex[0])
\t\tr = r<<4 | r
\t\tg = hexDigitToByte(hex[1])
\t\tg = g<<4 | g
\t\tb = hexDigitToByte(hex[2])
