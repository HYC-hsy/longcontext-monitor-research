// parseHexColor parses a hex color string and returns a color.Color.
// Supports 3, 4, 6, and 8 digit hex formats with or without '#' prefix.
func parseHexColor(hex string) (color.Color, error) {
\t// Remove '#' prefix if present
\thex = strings.TrimPrefix(hex, \"#\")

\tvar r, g, b, a uint8
\ta = 0xff // default alpha

\tswitch len(hex) {
\tcase 3:
\t\t// RGB: \"abc\" → 0xaa 0xbb 0xcc 0xff
\t\tvals, err := parseHexDigits(hex, 1)
\t\tif err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\t\tr = vals[0] | (vals[0] << 4)
\t\tg = vals[1] | (vals[1] << 4)
\t\tb = vals[2] | (vals[2] << 4)
\tcase 4:
\t\t// RGBA: \"abcd\" → 0xaa 0xbb 0xcc 0xdd
\t\tvals, err := parseHexDigits(hex, 1)
\t\tif err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\t\tr = vals[0] | (vals[0] << 4)
\t\tg = vals[1] | (vals[1] << 4)
\t\tb = vals[2] | (vals[2] << 4)
\t\ta = vals[3] | (vals[3] << 4)
\tcase 6:
\t\t// RRGGBB: \"a1b2c3\" → 0xa1 0xb2 0xc3 0xff
\t\tvals, err := parseHexDigits(hex, 2)
\t\tif err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\t\tr = vals[0]
\t\tg = vals[1]
\t\tb = vals[2]
\tcase 8:
\t\t// RRGGBBAA: \"a1b2c3f4\" → 0xa1 0xb2 0xc3 0xf4
\t\tvals, err := parseHexDigits(hex, 2)
\t\tif err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\t\tr = vals[0]
\t\tg = vals[1]
\t\tb = vals[2]
\t\ta = vals[3]
\tdefault:
\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color format: %s\", hex)
\t}

\treturn color.NRGBA{R: r, G: g, B: b, A: a}, nil
}
