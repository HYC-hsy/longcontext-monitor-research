// parseHexColor parses a hex color string in various formats:
// - 3 digits: \"abc\" -> #aabbcc
// - 4 digits: \"abcd\" -> #aabbccdd
// - 6 digits: \"a1b2c3\" -> #a1b2c3
// - 8 digits: \"a1b2c3f4\" -> #a1b2c3f4
// All formats support optional \"#\" prefix
func parseHexColor(hex string) (color.Color, error) {
\thex = strings.TrimPrefix(hex, \"#\")
\t
\tvar r, g, b, a uint8
\ta = 0xff // default alpha
\t
\tswitch len(hex) {
\tcase 3:
\t\t// RGB: each digit is doubled
\t\tvals, err := parseHexDigits(hex, 3)
\t\tif err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\t\tr = vals[0] | (vals[0] << 4)
\t\tg = vals[1] | (vals[1] << 4)
\t\tb = vals[2] | (vals[2] << 4)
\tcase 4:
\t\t// RGBA: each digit is doubled
\t\tvals, err := parseHexDigits(hex, 4)
\t\tif err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\t\tr = vals[0] | (vals[0] << 4)
\t\tg = vals[1] | (vals[1] << 4)
\t\tb = vals[2] | (vals[2] << 4)
\t\ta = vals[3] | (vals[3] << 4)
\tcase 6:
\t\t// RRGGBB
\t\tval, err := strconv.ParseUint(hex, 16, 32)
\t\tif err != nil {
\t\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color: %s\", hex)
\t\t}
\t\tr = uint8(val >> 16)
\t\tg = uint8(val >> 8)
\t\tb = uint8(val)
\tcase 8:
\t\t// RRGGBBAA
\t\tval, err := strconv.ParseUint(hex, 16, 32)
\t\tif err != nil {
\t\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color: %s\", hex)
\t\t}
\t\tr = uint8(val >> 24)
\t\tg = uint8(val >> 16)
\t\tb = uint8(val >> 8)
\t\ta = uint8(val)
\tdefault:
\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color length: %s\", hex)
\t}
\t
\treturn color.RGBA{R: r, G: g, B: b, A: a}, nil
}

// parseHexDigits parses n hex digits and returns their values

\t}
\t
\tvals := make([]uint8, n)
\tfor i := 0; i < n; i++ {
\t\tval, err := strconv.ParseUint(string(hex[i]), 16, 8)
\t\tif err != nil {
\t\t\treturn nil, fmt.Errorf(\"invalid hex digit: %c\", hex[i])
