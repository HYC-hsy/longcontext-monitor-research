// parseHexColor parses a hex color string and returns a color.Color.
// Supports formats: 3, 4, 6, and 8 digit hex (with or without # prefix).
func parseHexColor(hex string) (color.Color, error) {
\thex = strings.TrimPrefix(hex, \"#\")

\tvar r, g, b, a uint8
\ta = 0xff // default alpha

\tswitch len(hex) {
\tcase 3: // RGB
\t\tif _, err := fmt.Sscanf(hex, \"%1x%1x%1x\", &r, &g, &b); err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\t\tr = r*16 + r
\t\tg = g*16 + g
\t\tb = b*16 + b
\tcase 4: // RGBA
\t\tif _, err := fmt.Sscanf(hex, \"%1x%1x%1x%1x\", &r, &g, &b, &a); err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\t\tr = r*16 + r
\t\tg = g*16 + g
\t\tb = b*16 + b
\t\ta = a*16 + a
\tcase 6: // RRGGBB
\t\tif _, err := fmt.Sscanf(hex, \"%02x%02x%02x\", &r, &g, &b); err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\tcase 8: // RRGGBBAA
\t\tif _, err := fmt.Sscanf(hex, \"%02x%02x%02x%02x\", &r, &g, &b, &a); err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\tdefault:
\t\treturn color.Transparent, errors.New(\"invalid hex color format\")
\t}

\treturn color.NRGBA{R: r, G: g, B: b, A: a}, nil
}
