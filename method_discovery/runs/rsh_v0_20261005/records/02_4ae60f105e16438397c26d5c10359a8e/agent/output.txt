// parseHexColor parses a hex color string supporting 3, 4, 6, or 8 digit formats
// with or without the '#' prefix.
func parseHexColor(hex string) (color.Color, error) {
\t// Remove '#' prefix if present
\thex = strings.TrimPrefix(hex, \"#\")

\tvar r, g, b, a uint8
\ta = 0xff // default alpha

\tswitch len(hex) {
\tcase 3:
\t\t// RGB format: expand each digit (e.g., \"abc\" -> \"aabbcc\")
\t\tvals := make([]uint8, 3)
\t\tfor i := 0; i < 3; i++ {
\t\t\tv, err := strconv.ParseUint(string(hex[i]), 16, 8)
\t\t\tif err != nil {
\t\t\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color: %s\", hex)
\t\t\t}
\t\t\tvals[i] = uint8(v)
\t\t\tvals[i] = vals[i]<<4 | vals[i]
\t\t}
\t\tr, g, b = vals[0], vals[1], vals[2]

\tcase 4:
\t\t// RGBA format: expand each digit (e.g., \"abcd\" -> \"aabbccdd\")
\t\tvals := make([]uint8, 4)
\t\tfor i := 0; i < 4; i++ {
\t\t\tv, err := strconv.ParseUint(string(hex[i]), 16, 8)
\t\t\tif err != nil {
\t\t\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color: %s\", hex)
\t\t\t}
\t\t\tvals[i] = uint8(v)
\t\t\tvals[i] = vals[i]<<4 | vals[i]
\t\t}
\t\tr, g, b, a = vals[0], vals[1], vals[2], vals[3]

\tcase 6:
\t\t// RRGGBB format
\t\tv, err := strconv.ParseUint(hex, 16, 32)
\t\tif err != nil {
\t\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color: %s\", hex)
\t\t}
\t\tr = uint8(v >> 16)
\t\tg = uint8(v >> 8)
\t\tb = uint8(v)

\tcase 8:
\t\t// RRGGBBAA format
\t\tv, err := strconv.ParseUint(hex, 16, 64)
\t\tif err != nil {
\t\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color: %s\", hex)
\t\t}
\t\tr = uint8(v >> 24)
\t\tg = uint8(v >> 16)
\t\tb = uint8(v >> 8)
\t\ta = uint8(v)

\tdefault:
\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color length: %s\", hex)
\t}

\treturn color.NRGBA{R: r, G: g, B: b, A: a}, nil
}
