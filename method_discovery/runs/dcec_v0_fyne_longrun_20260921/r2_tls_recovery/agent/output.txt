// sprintfBinding is a String binding that formats multiple source bindings using sprintf

// NewSprintf creates a formatted string binding that combines multiple source bindings.
// The format string follows fmt.Sprintf conventions. When any source binding changes,
// the formatted string is automatically updated.
//
// The returned binding supports Set() which uses fmt.Sscanf to parse the string back
// into the source bindings. Note that Bytes bindings cannot be round-tripped.
//
// Since: 2.2
func NewSprintf(format string, b ...DataItem) String {
\tif len(b) == 0 {
\t\treturn NewString()
\t}

\tsprintf := &sprintfBinding{
\t\tformat:  format,
\t\tsources: b,
\t}

\t// Listen to all source bindings
\tfor _, source := range b {
\t\tsource.AddListener(NewDataListener(func() {
\t\t\tsprintf.trigger()
\t\t}))
\t}

\treturn sprintf
}

// Get returns the formatted string by reading all source bindings and applying sprintf
func (s *sprintfBinding) Get() (string, error) {
\tif s.lastErr != nil {
\t\treturn \"\", s.lastErr
