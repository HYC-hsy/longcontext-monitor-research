// parseHexColor parses a hex color string and returns a color.Color.
// Supports formats: 3-digit (#rgb), 4-digit (#rgba), 6-digit (#rrggbb), 8-digit (#rrggbbaa)
// The # prefix is optional.

\tvar r, g, b, a uint8
\ta = 0xff // default alpha

\tswitch len(hex) {

\t\t}
\t\tr = vals[0] | (vals[0] << 4)
\t\tg = vals[1] | (vals[1] << 4)
\t\tb = vals[2] | (vals[2] << 4)
\tcase 4: // rgba
\t\tvals, err := parseHexDigits(hex, 1)
\t\tif err != nil {
\t\t\treturn color.Transparent, err

\t\tg = vals[1] | (vals[1] << 4)

\t\ta = vals[3] | (vals[3] << 4)
\tcase 6: // rrggbb

\t\tif err != nil {

\t\t}
\t\tr = vals[0]
\t\tg = vals[1]

// parseHexDigits parses hex string into byte values with specified digit width (1 or 2)
func parseHexDigits(hex string, width int) ([]uint8, error) {
\tcount := len(hex) / width
\tresult := make([]uint8, count)
\t
\tfor i := 0; i < count; i++ {
\t\tstart := i * width
\t\tend := start + width
\t\tval, err := strconv.ParseUint(hex[start:end], 16, 8)
\t\tif err != nil {
\t\t\treturn nil, err
\t\t}
\t\tresult[i] = uint8(val)
\t}
\t
\treturn result, nil
}
