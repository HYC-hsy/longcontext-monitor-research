// FromJSON parses a JSON string and returns a Theme.
// On parse error, returns the default theme along with the error.
//
// Since: 2.5
func FromJSON(data string) (fyne.Theme, error) {
\treturn FromJSONReader(strings.NewReader(data))
}

// FromJSONReader parses JSON from an io.Reader and returns a Theme.
// On parse error, returns the default theme along with the error.
//
// Since: 2.5
func FromJSONReader(r io.Reader) (fyne.Theme, error) {
\tvar data jsonThemeData
\terr := json.NewDecoder(r).Decode(&data)
\tif err != nil {
\t\treturn DefaultTheme(), err
\t}

\ttheme := &jsonTheme{
\t\tcolors:      make(map[fyne.ThemeColorName]color.Color),
\t\tdarkColors:  make(map[fyne.ThemeColorName]color.Color),
\t\tlightColors: make(map[fyne.ThemeColorName]color.Color),
\t\tsizes:       make(map[fyne.ThemeSizeName]float32),
\t\tfonts:       make(map[string]fyne.Resource),
\t\ticons:       make(map[fyne.ThemeIconName]fyne.Resource),
\t}

\t// Parse colors
\tfor name, hexStr := range data.Colors {
\t\tc, err := parseHexColor(hexStr)
\t\tif err == nil {
\t\t\ttheme.colors[fyne.ThemeColorName(name)] = c
\t\t}
\t}

\t// Parse dark variant colors
\tfor name, hexStr := range data.ColorsDark {
\t\tc, err := parseHexColor(hexStr)
\t\tif err == nil {
\t\t\ttheme.darkColors[fyne.ThemeColorName(name)] = c
\t\t}
\t}

\t// Parse light variant colors
\tfor name, hexStr := range data.ColorsLight {
\t\tc, err := parseHexColor(hexStr)
\t\tif err == nil {
\t\t\ttheme.lightColors[fyne.ThemeColorName(name)] = c
\t\t}
\t}

\t// Parse sizes
\tfor name, size := range data.Sizes {
\t\ttheme.sizes[fyne.ThemeSizeName(name)] = size
\t}

\t// Parse fonts
\tfor style, uri := range data.Fonts {
\t\tif resource, err := loadResource(uri); err == nil {
\t\t\ttheme.fonts[style] = resource
\t\t}
\t}

\t// Parse icons
\tfor name, uri := range data.Icons {
\t\tif resource, err := loadResource(uri); err == nil {
\t\t\ttheme.icons[fyne.ThemeIconName(name)] = resource
\t\t}
\t}

\treturn theme, nil
}

func (t *jsonTheme) Color(name fyne.ThemeColorName, variant fyne.ThemeVariant) color.Color {
\t// Check variant-specific colors first
\tif variant == fyne.VariantDark {
\t\tif c, ok := t.darkColors[name]; ok {
\t\t\treturn c
\t\t}
\t} else if variant == fyne.VariantLight {
\t\tif c, ok := t.lightColors[name]; ok {
\t\t\treturn c
\t\t}
\t}

\t// Check generic colors
\tif c, ok := t.colors[name]; ok {
\t\treturn c
\t}

\t// Fall back to default theme
\treturn DefaultTheme().Color(name, variant)
}

func (t *jsonTheme) Font(style fyne.TextStyle) fyne.Resource {
\tvar key string
\tif style.Monospace {
\t\tkey = \"monospace\"
\t} else if style.Bold && style.Italic {
\t\tkey = \"boldItalic\"
\t} else if style.Bold {
\t\tkey = \"bold\"
\t} else if style.Italic {
\t\tkey = \"italic\"
\t} else {
\t\tkey = \"regular\"
\t}

\tif resource, ok := t.fonts[key]; ok {
\t\treturn resource
\t}

\treturn DefaultTheme().Font(style)
}

func (t *jsonTheme) Icon(name fyne.ThemeIconName) fyne.Resource {
\tif resource, ok := t.icons[name]; ok {
\t\treturn resource
\t}

\treturn DefaultTheme().Icon(name)
}

func (t *jsonTheme) Size(name fyne.ThemeSizeName) float32 {
\tif size, ok := t.sizes[name]; ok {
\t\treturn size
\t}

\treturn DefaultTheme().Size(name)
}
