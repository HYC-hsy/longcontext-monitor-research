// parseHexColor parses a hex color string in various formats:
// - 3 digits: \"abc\" or \"#abc\" -> #aabbcc
// - 4 digits: \"abcd\" or \"#abcd\" -> #aabbccdd
// - 6 digits: \"a1b2c3\" or \"#a1b2c3\" -> #a1b2c3
// - 8 digits: \"a1b2c3f4\" or \"#a1b2c3f4\" -> #a1b2c3f4
func parseHexColor(hex string) (color.Color, error) {
\t// Remove # prefix if present
\thex = strings.TrimPrefix(hex, \"#\")

\tvar r, g, b, a uint8
\ta = 0xff // default to opaque

\tswitch len(hex) {
\tcase 3:
\t\t// RGB shorthand: abc -> aabbcc
\t\tvals, err := parseHexDigits(hex)
\t\tif err != nil {
\t\t\treturn color.NRGBA{}, err
\t\t}
\t\tr = vals[0] | (vals[0] << 4)
\t\tg = vals[1] | (vals[1] << 4)
\t\tb = vals[2] | (vals[2] << 4)

\tcase 4:
\t\t// RGBA shorthand: abcd -> aabbccdd
\t\tvals, err := parseHexDigits(hex)
\t\tif err != nil {
\t\t\treturn color.NRGBA{}, err
\t\t}
\t\tr = vals[0] | (vals[0] << 4)
\t\tg = vals[1] | (vals[1] << 4)
\t\tb = vals[2] | (vals[2] << 4)
\t\ta = vals[3] | (vals[3] << 4)

\tcase 6:
\t\t// RRGGBB
\t\tval, err := strconv.ParseUint(hex, 16, 32)
\t\tif err != nil {
\t\t\treturn color.NRGBA{}, err
\t\t}
\t\tr = uint8((val >> 16) & 0xff)
\t\tg = uint8((val >> 8) & 0xff)
\t\tb = uint8(val & 0xff)

\tcase 8:
\t\t// RRGGBBAA
\t\tval, err := strconv.ParseUint(hex, 16, 64)
\t\tif err != nil {
\t\t\treturn color.NRGBA{}, err
\t\t}
\t\tr = uint8((val >> 24) & 0xff)
\t\tg = uint8((val >> 16) & 0xff)
\t\tb = uint8((val >> 8) & 0xff)
\t\ta = uint8(val & 0xff)

\tdefault:
\t\treturn color.NRGBA{}, fmt.Errorf(\"invalid hex color format: %s\", hex)
\t}

\treturn color.NRGBA{R: r, G: g, B: b, A: a}, nil
}

\t\t\treturn nil, fmt.Errorf(\"invalid hex digit: %c\", ch)
\t\t}
\t\tvals[i] = val
\t}
\treturn vals, nil
}
