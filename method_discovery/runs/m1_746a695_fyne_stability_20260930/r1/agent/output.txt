// parseHexColor parses a hex color string in various formats (3, 4, 6, or 8 digits, with or without # prefix).
func parseHexColor(hex string) (color.Color, error) {
\t// Remove # prefix if present
\thex = strings.TrimPrefix(hex, \"#\")

\t// Validate hex characters
\tfor _, c := range hex {
\t\tif !((c >= '0' && c <= '9') || (c >= 'a' && c <= 'f') || (c >= 'A' && c <= 'F')) {
\t\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color format: %s\", hex)
\t\t}
\t}

\tvar r, g, b, a uint8
\ta = 0xff // Default alpha to fully opaque

\tswitch len(hex) {
\tcase 3:
\t\t// RGB format: \"abc\" -> 0xaa 0xbb 0xcc 0xff
\t\trv, _ := strconv.ParseUint(string(hex[0]), 16, 8)
\t\tgv, _ := strconv.ParseUint(string(hex[1]), 16, 8)
\t\tbv, _ := strconv.ParseUint(string(hex[2]), 16, 8)
\t\tr, g, b = uint8(rv*17), uint8(gv*17), uint8(bv*17)

\tcase 4:
\t\t// RGBA format: \"abcd\" -> 0xaa 0xbb 0xcc 0xdd
\t\trv, _ := strconv.ParseUint(string(hex[0]), 16, 8)
\t\tgv, _ := strconv.ParseUint(string(hex[1]), 16, 8)
\t\tbv, _ := strconv.ParseUint(string(hex[2]), 16, 8)
\t\tav, _ := strconv.ParseUint(string(hex[3]), 16, 8)
\t\tr, g, b, a = uint8(rv*17), uint8(gv*17), uint8(bv*17), uint8(av*17)

\tcase 6:
\t\t// RRGGBB format: \"a1b2c3\" -> 0xa1 0xb2 0xc3 0xff
\t\tval, _ := strconv.ParseUint(hex, 16, 32)
\t\tr = uint8((val >> 16) & 0xff)
\t\tg = uint8((val >> 8) & 0xff)
\t\tb = uint8(val & 0xff)

\tcase 8:
\t\t// RRGGBBAA format: \"a1b2c3f4\" -> 0xa1 0xb2 0xc3 0xf4
\t\tval, _ := strconv.ParseUint(hex, 16, 64)
\t\tr = uint8((val >> 24) & 0xff)
\t\tg = uint8((val >> 16) & 0xff)
\t\tb = uint8((val >> 8) & 0xff)
\t\ta = uint8(val & 0xff)

\tdefault:
\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color length: %d\", len(hex))
\t}

\treturn color.NRGBA{R: r, G: g, B: b, A: a}, nil
}
