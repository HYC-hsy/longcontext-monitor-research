// parseHexColor parses hex color strings in various formats:
// - 3 digits: \"abc\" or \"#abc\" -> #aabbcc
// - 4 digits: \"abcd\" or \"#abcd\" -> #aabbccdd  
// - 6 digits: \"a1b2c3\" or \"#a1b2c3\" -> #a1b2c3
// - 8 digits: \"a1b2c3f4\" or \"#a1b2c3f4\" -> #a1b2c3f4
func parseHexColor(s string) (color.Color, error) {
\ts = strings.TrimPrefix(s, \"#\")
\t
\tvar r, g, b, a uint8 = 0, 0, 0, 255
\t
\tswitch len(s) {
\tcase 3:
\t\t// RGB shorthand: abc -> aabbcc
\t\tvals, err := parseHexDigits(s, 3)
\t\tif err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\t\tr = vals[0] * 17 // 0xa -> 0xaa
\t\tg = vals[1] * 17
\t\tb = vals[2] * 17
\tcase 4:
\t\t// RGBA shorthand: abcd -> aabbccdd
\t\tvals, err := parseHexDigits(s, 4)
\t\tif err != nil {
\t\t\treturn color.Transparent, err
\t\t}
\t\tr = vals[0] * 17
\t\tg = vals[1] * 17
\t\tb = vals[2] * 17
\t\ta = vals[3] * 17
\tcase 6:
\t\t// RRGGBB
\t\tval, err := strconv.ParseUint(s, 16, 32)
\t\tif err != nil {
\t\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color: %s\", s)
\t\t}
\t\tr = uint8(val >> 16)
\t\tg = uint8(val >> 8)
\t\tb = uint8(val)
\tcase 8:
\t\t// RRGGBBAA
\t\tval, err := strconv.ParseUint(s, 16, 64)
\t\tif err != nil {
\t\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color: %s\", s)
\t\t}
\t\tr = uint8(val >> 24)
\t\tg = uint8(val >> 16)
\t\tb = uint8(val >> 8)
\t\ta = uint8(val)
\tdefault:
\t\treturn color.Transparent, fmt.Errorf(\"invalid hex color length: %s\", s)
\t}
\t
\treturn color.NRGBA{R: r, G: g, B: b, A: a}, nil
}

}
