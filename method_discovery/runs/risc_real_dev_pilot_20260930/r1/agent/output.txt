// FromJSON parses a JSON string and returns a Theme that applies overrides from the JSON.
// Colors, sizes, fonts, and icons not specified in the JSON fall back to the default theme.
// The JSON structure supports:
// - \"Colors\": map of color names to hex strings (applied to both variants)
// - \"Colors-dark\": map of color names to hex strings (applied only to dark variant)
// - \"Colors-light\": map of color names to hex strings (applied only to light variant)
// - \"Sizes\": map of size names to float32 values
// - \"Fonts\": map of font style names (\"regular\", \"bold\", \"italic\", \"boldItalic\", \"monospace\") to URI strings
// - \"Icons\": map of icon names to URI strings
//
// Since: 2.2
func FromJSON(data string) (fyne.Theme, error) {
\treturn FromJSONReader(strings.NewReader(data))
}

// FromJSONReader reads from an io.Reader and returns a Theme that applies overrides from the JSON.
// On parse error, returns the default theme along with the error.
//
// Since: 2.2
func FromJSONReader(r io.Reader) (fyne.Theme, error) {
\tvar raw map[string]interface{}
\tdecoder := json.NewDecoder(r)
\terr := decoder.Decode(&raw)
\tif err != nil {
\t\treturn DefaultTheme(), err
\t}

\ttheme := &jsonTheme{
\t\tcolors:      make(map[string]map[fyne.ThemeColorName]color.Color),
\t\tsizes:       make(map[string]float32),
\t\tfonts:       make(map[string]fyne.Resource),
\t\ticons:       make(map[fyne.ThemeIconName]fyne.Resource),
\t\tdefaultTheme: DefaultTheme(),
\t}

\t// Parse generic colors
\tif colorsRaw, ok := raw[\"Colors\"].(map[string]interface{}); ok {

\t}

\t// Parse variant-specific colors

\t}

func parseColorMap(raw map[string]interface{}) map[fyne.ThemeColorName]color.Color {
\tcolors := make(map[fyne.ThemeColorName]color.Color)
\tfor name, hexVal := range raw {
\t\tif hexStr, ok := hexVal.(string); ok {
\t\t\tif c, err := parseHexColor(hexStr); err == nil {
\t\t\t\tcolors[fyne.ThemeColorName(name)] = c
\t\t\t}
\t\t}
\t}
\treturn colors
}

func parseHexColor(hex string) (color.Color, error) {
\t// Remove # prefix if present
\thex = strings.TrimPrefix(hex, \"#\")

\tvar r, g, b, a uint8
\ta = 0xff // default alpha

\tswitch len(hex) {
\tcase 3:
\t\t// RGB shorthand: abc -> aabbcc
\t\tif val, err := strconv.ParseUint(hex, 16, 16); err == nil {
\t\t\tr = uint8((val >> 8) & 0xf)
\t\t\tr = r<<4 | r
\t\t\tg = uint8((val >> 4) & 0xf)
\t\t\tg = g<<4 | g
\t\t\tb = uint8(val & 0xf)
\t\t\tb = b<<4 | b
\t\t} else {
\t\t\treturn color.Transparent, err
\t\t}
