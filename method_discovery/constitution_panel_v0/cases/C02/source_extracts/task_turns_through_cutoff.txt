        }
    }

impl<'a> StatefulWidget for Scrollbar<'a> {
    type State = ScrollbarState;

    fn render(self, area: Rect, buf: &mut Buffer, state: &mut Self::State) {
        if state.content_length == 0 {
            return;
        }

        let (scrollbar_area, is_vertical) = match self.orientation {
            ScrollbarOrientation::VerticalRight => {
                if area.width == 0 {
                    return;
                }
                (
                    Rect {
                        x: area.x + area.width - 1,
                        y: area.y,
                        width: 1,
                        height: area.height,
                    },
                    true,
                )
            }
            ScrollbarOrientation::VerticalLeft => {
                if area.width == 0 {
                    return;
                }
                (
                    Rect {
                        x: area.x,
                        y: area.y,
                        width: 1,
                        height: area.height,
                    },
                    true,
                )
            }
            ScrollbarOrientation::HorizontalBottom => {
                if area.height == 0 {
                    return;
                }
                (
                    Rect {
                        x: area.x,
                        y: area.y + area.height - 1,
                        width: area.width,
                        height: 1,
                    },
                    false,
                )
            }
            ScrollbarOrientation::HorizontalTop => {
                if area.height == 0 {
                    return;
                }
                (
                    Rect {
                        x: area.x,
                        y: area.y,
                        width: area.width,
                        height: 1,
                    },
                    false,
                )
            }
        };

        let track_length = if is_vertical {
            scrollbar_area.height
        } else {
            scrollbar_area.width
        } as usize;

        if track_length == 0 {
            return;
        }

        // Calculate track length accounting for begin/end symbols
        let mut track_start = 0;
        let mut track_end = track_length;

        if self.begin_symbol.is_some() {
            track_start += 1;
        }
        if self.end_symbol.is_some() {
            track_end = track_end.saturating_sub(1);
        }

        let available_track = track_end.saturating_sub(track_start);
        if available_track < 1 {
            return;
        }

        // Calculate thumb size and position
        let thumb_size = if state.viewport_content_length == 0 {
            available_track
        } else {
            let ratio = state.viewport_content_length as f64 / state.content_length as f64;
            (available_track as f64 * ratio).round().max(1.0) as usize
        }
        .min(available_track);

        let scrollable_area = state.content_length.saturating_sub(state.viewport_content_length);
        let thumb_position = if scrollable_area == 0 {
            0
        } else {
            let ratio = state.position as f64 / scrollable_area as f64;
            let max_thumb_pos = available_track.saturating_sub(thumb_size);
            (max_thumb_pos as f64 * ratio).round() as usize
        };

        // Render the scrollbar
        for i in 0..track_length {
            let (x, y) = if is_vertical {
                (scrollbar_area.x, scrollbar_area.y + i as u16)
            } else {
                (scrollbar_area.x + i as u16, scrollbar_area.y)
            };

            let (symbol, style) = if i == 0 && self.begin_symbol.is_some() {
                (self.begin_symbol.unwrap(), self.begin_style)
            } else if i == track_length - 1 && self.end_symbol.is_some() {
                (self.end_symbol.unwrap(), self.end_style)
            } else {
                let track_pos = i.saturating_sub(track_start);
                if track_pos >= thumb_position && track_pos < thumb_position + thumb_size {
                    (self.thumb_symbol, self.thumb_style)
                } else {
                    (self.track_symbol, self.track_style)
                }
            };

            buf.get_mut(x, y)
                .set_symbol(symbol)
                .set_style(self.style.patch(style));
        }
    }
}

}
