namespace glz
{
   // Forward declarations
   class lazy_json_view;
   class indexed_lazy_view;
   template <auto Opts = opts{}>
   class lazy_document;

   namespace detail
   {
      // JSON value type enumeration
      enum class json_type : uint8_t { null, boolean, number, string, array, object, error };

      // Determine JSON type from first character (after whitespace)
      inline json_type detect_json_type(char c) noexcept
      {
         switch (c) {
         case 'n':
            return json_type::null;
         case 't':
         case 'f':
            return json_type::boolean;
         case '\"':
            return json_type::string;
         case '[':
            return json_type::array;
         case '{':
            return json_type::object;
         case '-':
         case '0':
         case '1':
         case '2':
         case '3':
         case '4':
         case '5':
         case '6':
         case '7':
         case '8':
         case '9':
            return json_type::number;
         default:
            return json_type::error;
         }
      }

      // Find the end of a JSON value starting at 'it', returning the position after the value
      template <auto Opts>
      inline std::string_view::iterator find_value_end(std::string_view::iterator it,
                                                         std::string_view::iterator end) noexcept
      {
         context ctx{};
         auto start = it;
         skip_value<JSON>::op<Opts>(ctx, it, end);
         if (bool(ctx.error)) {
            return start; // Return start on error
         }
         return it;
      }

      // Skip whitespace and return the first non-whitespace character position
      template <auto Opts>
      inline std::string_view::iterator skip_whitespace(std::string_view::iterator it,
                                                          std::string_view::iterator end) noexcept
      {
         if constexpr (Opts.minified) {
            return it; // No whitespace in minified JSON
         }
         else {
            while (it != end && (*it == ' ' || *it == '\\t' || *it == '\
' || *it == '\\r')) {
               ++it;
            }
            return it;
         }
      }
   } // namespace detail

   // Lazy JSON view - represents a view into a JSON value without parsing
   class lazy_json_view
   {
     private:
      std::string_view buffer_; // The entire JSON buffer
      std::string_view value_; // The specific value range
      detail::json_type type_ = detail::json_type::error;
      bool error_ = false;

      template <auto Opts>
      friend class lazy_document;
      friend class indexed_lazy_view;

      // Private constructor for creating views
      lazy_json_view(std::string_view buffer, std::string_view value, detail::json_type type, bool error = false)
         : buffer_(buffer), value_(value), type_(type), error_(error)
      {}

     public:
      // Default constructor creates an error view
      lazy_json_view() : error_(true) {}

      // Type predicates
      [[nodiscard]] bool is_object() const noexcept { return !error_ && type_ == detail::json_type::object; }
      [[nodiscard]] bool is_array() const noexcept { return !error_ && type_ == detail::json_type::array; }
      [[nodiscard]] bool is_string() const noexcept { return !error_ && type_ == detail::json_type::string; }
      [[nodiscard]] bool is_number() const noexcept { return !error_ && type_ == detail::json_type::number; }
      [[nodiscard]] bool is_boolean() const noexcept { return !error_ && type_ == detail::json_type::boolean; }
      [[nodiscard]] bool is_null() const noexcept { return !error_ && type_ == detail::json_type::null; }

      // Error checking
      [[nodiscard]] bool has_error() const noexcept { return error_; }

      // Get the raw JSON string for this value
      [[nodiscard]] std::string_view raw_json() const noexcept { return value_; }

      // Extract typed values
      template <typename T>
      [[nodiscard]] std::optional<T> get() const noexcept
      {
         if (error_) {
            return std::nullopt;
         }

         T result{};
         context ctx{};
         auto it = value_.begin();
         auto end = value_.end();

         // Use the standard JSON reader
         if (read<JSON>::op<opts{}>(result, ctx, it, end) != error_code::none) {
            return std::nullopt;
         }

         return result;
      }

      // Direct deserialization into a value
      template <typename T>
      [[nodiscard]] error_code read_into(T& value) const noexcept
      {
         if (error_) {
            return error_code::parse_error;
         }

         context ctx{};
         auto it = value_.begin();
         auto end = value_.end();
         return read<JSON>::op<opts{}>(value, ctx, it, end);
      }

      // Navigate to object member by key
      template <auto Opts = opts{}>
      [[nodiscard]] lazy_json_view operator[](std::string_view key) const noexcept
      {
         if (error_ || type_ != detail::json_type::object) {
            return lazy_json_view{}; // Return error view
         }

         auto it = value_.begin();
         auto end = value_.end();

         // Skip opening brace
         ++it;
         if (it == end) {
            return lazy_json_view{};
         }

         it = detail::skip_whitespace<Opts>(it, end);

         // Handle empty object
         if (it != end && *it == '}') {
            return lazy_json_view{}; // Key not found
         }

         // Search for the key
         while (it != end) {
            // Skip whitespace before key
            it = detail::skip_whitespace<Opts>(it, end);
            if (it == end || *it != '\"') {
               return lazy_json_view{};
            }

            // Parse the key
            auto key_start = it;
            ++it; // Skip opening quote
            while (it != end && *it != '\"') {
               if (*it == '\\\\') {
                  ++it; // Skip escape character
                  if (it == end) return lazy_json_view{};
               }
               ++it;
            }
            if (it == end) {
               return lazy_json_view{};
            }

            auto key_end = it;
            ++it; // Skip closing quote

            // Extract key (including quotes for now, will need to decode if necessary)
            std::string_view found_key(key_start + 1, key_end - key_start - 1);

            // Skip whitespace and colon
            it = detail::skip_whitespace<Opts>(it, end);
            if (it == end || *it != ':') {
               return lazy_json_view{};
            }
            ++it; // Skip colon
            it = detail::skip_whitespace<Opts>(it, end);

            // Get value range
            auto value_start = it;
            auto value_end = detail::find_value_end<Opts>(it, end);
            if (value_start == value_end) {
               return lazy_json_view{};
            }

            // Check if this is the key we're looking for
            if (found_key == key) {
               std::string_view value_range(value_start, value_end - value_start);
               auto value_type = detail::detect_json_type(*value_start);
               return lazy_json_view(buffer_, value_range, value_type, false);
            }

            // Skip to next key
            it = value_end;
            it = detail::skip_whitespace<Opts>(it, end);
            if (it == end) {
               return lazy_json_view{};
            }
            if (*it == '}') {
               return lazy_json_view{}; // Key not found
            }
            if (*it == ',') {
               ++it; // Skip comma
            }
            else {
               return lazy_json_view{}; // Syntax error
            }
         }

         return lazy_json_view{}; // Key not found
      }

      // Navigate to array element by index
      template <auto Opts = opts{}>
      [[nodiscard]] lazy_json_view operator[](size_t index) const noexcept
      {
         if (error_ || type_ != detail::json_type::array) {
            return lazy_json_view{}; // Return error view
         }

         auto it = value_.begin();
         auto end = value_.end();

         // Skip opening bracket
         ++it;
         if (it == end) {
            return lazy_json_view{};
         }

         it = detail::skip_whitespace<Opts>(it, end);

         // Handle empty array
         if (it != end && *it == ']') {
            return lazy_json_view{}; // Index out of bounds
         }

         // Skip to the desired index
         size_t current_index = 0;
         while (it != end) {
            it = detail::skip_whitespace<Opts>(it, end);
            if (it == end) {
               return lazy_json_view{};
            }

            auto value_start = it;
            auto value_end = detail::find_value_end<Opts>(it, end);
            if (value_start == value_end) {
               return lazy_json_view{};
            }

            if (current_index == index) {
               std::string_view value_range(value_start, value_end - value_start);
               auto value_type = detail::detect_json_type(*value_start);
               return lazy_json_view(buffer_, value_range, value_type, false);
            }

            ++current_index;
            it = value_end;
            it = detail::skip_whitespace<Opts>(it, end);
            if (it == end) {
               return lazy_json_view{};
            }
            if (*it == ']') {
               return lazy_json_view{}; // Index out of bounds
            }
            if (*it == ',') {
               ++it; // Skip comma
            }
            else {
               return lazy_json_view{}; // Syntax error
            }
         }

         return lazy_json_view{}; // Index out of bounds

            auto it = value_.begin();
            auto end = value_.end();

            if (it == end) return 0;

            if (it != end && *it == '}') return 0; // Empty object

               // Skip key (must be a string)
               if (*it != '\"') return count;
               ++it;
               while (it != end && *it != '\"') {
                  if (*it == '\\\\') ++it;
                  if (it != end) ++it;
               }
               if (it == end) return count;
               ++it; // Skip closing quote

               // Skip colon
               it = detail::skip_whitespace<Opts>(it, end);
               if (it == end || *it != ':') return count;
               ++it;
               it = detail::skip_whitespace<Opts>(it, end);

               // Skip value
               it = detail::find_value_end<Opts>(it, end);
               ++count;

               it = detail::skip_whitespace<Opts>(it, end);
               if (it == end || *it == '}') break;
               if (*it == ',') ++it;
               else return count;

               it = detail::skip_whitespace<Opts>(it, end);
            }
            return count;
         }

            size_t count = 0;
            auto it = value_.begin();
            auto end = value_.end();

            if (it == end) return 0;

            if (it != end && *it == ']') return 0; // Empty array

               it = detail::find_value_end<Opts>(it, end);
               ++count;

               it = detail::skip_whitespace<Opts>(it, end);
               if (it == end || *it == ']') break;
               if (*it == ',') ++it;
               else return count;

               it = detail::skip_whitespace<Opts>(it, end);
