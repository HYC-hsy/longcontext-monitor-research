\t\t\tg = uint8((val >> 4) & 0xf)

// NewSprintf creates a formatted string binding that combines multiple data bindings.
// The format string works like fmt.Sprintf, and the binding updates automatically
// when any source binding changes.
//
// Since: 2.5
func NewSprintf(format string, bindings ...DataItem) String {
\tsb := &sprintfBinding{
\t\tformat:  format,
\t\tsources: bindings,
\t}

\t// Listen to all source bindings
\tlistener := NewDataListener(func() {
\t\tsb.trigger()
\t})

\tfor _, b := range bindings {
\t\tb.AddListener(listener)
\t}

\treturn sb
}

func (sb *sprintfBinding) Get() (string, error) {
\tif sb.lastErr != nil {
\t\treturn \"\", sb.lastErr
\t}

\tvalues := make([]interface{}, len(sb.sources))
\tfor i, source := range sb.sources {
\t\tval, err := sb.getValue(source)
\t\tif err != nil {
\t\t\treturn \"\", err
\t\t}
\t\tvalues[i] = val
\t}

\treturn fmt.Sprintf(sb.format, values...), nil
}

func (sb *sprintfBinding) Set(str string) error {
\tvalues := make([]interface{}, len(sb.sources))
\t
\t// Prepare pointers for Sscanf
\tfor i, source := range sb.sources {
\t\tswitch source.(type) {
\t\tcase Bool:
\t\t\tvar v bool
\t\t\tvalues[i] = &v
\t\tcase Int:
\t\t\tvar v int
\t\t\tvalues[i] = &v
\t\tcase Float:
\t\t\tvar v float64
\t\t\tvalues[i] = &v
\t\tcase String:
\t\t\tvar v string
\t\t\tvalues[i] = &v
\t\tcase Rune:
\t\t\tvar v rune
\t\t\tvalues[i] = &v
\t\tcase Bytes:
\t\t\t// Bytes cannot be round-tripped
\t\t\treturn fmt.Errorf(\"cannot parse Bytes binding from string\")
\t\tcase URI:
\t\t\tvar v string
\t\t\tvalues[i] = &v
\t\tdefault:
\t\t\treturn fmt.Errorf(\"unsupported binding type\")
\t\t}
\t}

\t// Parse the string
\tn, err := fmt.Sscanf(str, sb.format, values...)
\tif err != nil {
\t\treturn err
\t}
\tif n != len(sb.sources) {
\t\treturn fmt.Errorf(\"parsed %d values, expected %d\", n, len(sb.sources))
\t}

\t// Set the parsed values back to source bindings
\tfor i, source := range sb.sources {
\t\tif err := sb.setValue(source, values[i]); err != nil {
\t\t\treturn err
\t\t}
\t}

\treturn nil
