\t\"sync\"

var (
\tappMetadata      fyne.AppMetadata
\tappMetadataLock  sync.RWMutex
\tmetadataSet      bool
)

// SetMetadata allows an application to override its build metadata at runtime.
// This sets the global metadata that will be returned by any App's Metadata() method.
func SetMetadata(m fyne.AppMetadata) {
\tappMetadataLock.Lock()
\tdefer appMetadataLock.Unlock()

\tappMetadata = m
\tmetadataSet = true
}

func getMetadata() fyne.AppMetadata {
\tappMetadataLock.RLock()
\tdefer appMetadataLock.RUnlock()

\tif metadataSet {
\t\treturn appMetadata
\t}

\t// Return default metadata if none has been set
\treturn fyne.AppMetadata{
\t\tID:      \"com.example\",
\t\tName:    \"Fyne App\",
\t\tVersion: \"1.0.0\",
\t\tBuild:   1,
\t}
}
