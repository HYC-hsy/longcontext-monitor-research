{
  "path": "/app/program.py",
  "content": "#!/usr/bin/env python3
\"\"\"
Python reimplementation of COBOL BOOKFORUM program.
Reads INPUT.DAT and processes book transaction, updating ACCOUNTS.DAT, 
BOOKS.DAT, and TRANSACTIONS.DAT files.
\"\"\"

import sys
import os

# Record structure sizes (in bytes)
ACCOUNT_RECORD_SIZE = 34  # 4 + 20 + 10
BOOK_RECORD_SIZE = 28     # 4 + 20 + 4
TRANSACTION_RECORD_SIZE = 22  # 4 + 10 + 4 + 4


def read_input_file(filepath):
    \"\"\"Read and parse INPUT.DAT file.\"\"\"
    with open(filepath, 'rb') as f:
        data = f.read()
    
    # INPUT.DAT has only 15 bytes, but COBOL expects 22
    # Pad with spaces to match COBOL behavior
    data = data.ljust(22, b' ')
    
    # Parse fields according to COBOL structure
    # 01 INPUT-RECORD.
    #   05 INPUT-BUYER-ID   PIC X(4).
    #   05 INPUT-SELLER-ID  PIC X(4).
    #   05 INPUT-BOOK-ID    PIC X(4).
    #   05 INPUT-AMOUNT     PIC 9(10).
    
    buyer_id = data[0:4].decode('ascii')
    seller_id = data[4:8].decode('ascii')
    book_id = data[8:12].decode('ascii')
    amount_str = data[12:22].decode('ascii')
    
    # Convert amount string to integer
    # COBOL PIC 9(10) is numeric, spaces are treated as zeros
    amount_str = amount_str.replace(' ', '0')
    amount = int(amount_str)
    
    return buyer_id, seller_id, book_id, amount


def read_accounts_file(filepath):
    \"\"\"Read all account records from ACCOUNTS.DAT.\"\"\"
    accounts = []
    with open(filepath, 'rb') as f:
        while True:
            record = f.read(ACCOUNT_RECORD_SIZE)
            if not record:
                break
            
            # Parse ACCOUNT-RECORD structure
            account_id = record[0:4].decode('ascii')
            account_name = record[4:24].decode('ascii')
            account_balance = int(record[24:34].decode('ascii'))
            
            accounts.append({
                'id': account_id,
                'name': account_name,
                'balance': account_balance
            })
    
    return accounts


def write_accounts_file(filepath, accounts):
    \"\"\"Write account records back to ACCOUNTS.DAT.\"\"\"
    with open(filepath, 'wb') as f:
        for account in accounts:
            # Format: ACCOUNT-ID (4) + ACCOUNT-NAME (20) + ACCOUNT-BALANCE (10)
            record = (account['id'].ljust(4)[:4] +
                     account['name'].ljust(20)[:20] +
                     str(account['balance']).zfill(10))
            f.write(record.encode('ascii'))


def read_books_file(filepath):
    \"\"\"Read all book records from BOOKS.DAT.\"\"\"
    books = []
    with open(filepath, 'rb') as f:
        while True:
            record = f.read(BOOK_RECORD_SIZE)
            if not record:
                break
            
            # Parse BOOK-RECORD structure
            book_id = record[0:4].decode('ascii')
            book_title = record[4:24].decode('ascii')
            book_owner = record[24:28].decode('ascii')
            
            books.append({
                'id': book_id,
                'title': book_title,
                'owner': book_owner
            })
    
    return books


def write_books_file(filepath, books):
    \"\"\"Write book records back to BOOKS.DAT.\"\"\"
    with open(filepath, 'wb') as f:
        for book in books:
            # Format: BOOK-ID (4) + BOOK-TITLE (20) + BOOK-OWNER (4)
            record = (book['id'].ljust(4)[:4] +
                     book['title'].ljust(20)[:20] +
                     book['owner'].ljust(4)[:4])
            f.write(record.encode('ascii'))


def append_transaction(filepath, book_id, amount, seller_id, buyer_id):
    \"\"\"Append a transaction record to TRANSACTIONS.DAT.\"\"\"
    # Format: TRANS-BOOK (4) + TRANS-AMOUNT (10) + TRANS-SELLER (4) + TRANS-BUYER (4)
    record = (book_id.ljust(4)[:4] +
             str(amount).zfill(10) +
             seller_id.ljust(4)[:4] +
             buyer_id.ljust(4)[:4])
    
    with open(filepath, 'ab') as f:
        f.write(record.encode('ascii'))


def main():
    \"\"\"Main program logic matching COBOL BOOKFORUM.\"\"\"
    
    # File paths
    input_file = 'src/INPUT.DAT'
    accounts_file = 'data/ACCOUNTS.DAT'
    books_file = 'data/BOOKS.DAT'
    transactions_file = 'data/TRANSACTIONS.DAT'
    
    # Read input file
    try:
        buyer_id, seller_id, book_id, amount = read_input_file(input_file)
    except Exception as e:
        print(\"Error: Input file is empty\")
        sys.exit(1)
    
    # VALIDATE-DATA section
    # Check if buyer exists
    accounts = read_accounts_file(accounts_file)
    buyer_found = False
    seller_found = False
    
    for account in accounts:
        if account['id'] == buyer_id:
            buyer_found = True
        if account['id'] == seller_id:
            seller_found = True
    
    # Check if seller exists and owns the book
    books = read_books_file(books_file)
    book_found = False
    valid_owner = False
    
    for book in books:
        if book['id'] == book_id:
            book_found = True
            if book['owner'] == seller_id:
                valid_owner = True
    
    # Display empty line (matches COBOL line 119)
    print(\" \")
    
    # Validate all conditions
    if not buyer_found:
        print(\"Error: Buyer account not found\")
        sys.exit(1)
    
    if not seller_found:
        print(\"Error: Seller account not found\")
        sys.exit(1)
    
    if not book_found:
        print(\"Error: Book not found\")
        sys.exit(1)
    
    if not valid_owner:
        print(\"Error: Seller does not own the book\")
        sys.exit(1)
    
    # PROCESS-TRANSACTION section
    # Update account balances
    for account in accounts:
        if account['id'] == buyer_id:
            account['balance'] -= amount
        if account['id'] == seller_id:
            account['balance'] += amount
    
    write_accounts_file(accounts_file, accounts)
    
    # Transfer book ownership
    for book in books:
        if book['id'] == book_id:
            book['owner'] = buyer_id
    
    write_books_file(books_file, books)
    
    # Append transaction record
    append_transaction(transactions_file, book_id, amount, seller_id, buyer_id)
    
    print(\"Transaction completed successfully\")


if __name__ == '__main__':
    main()
"
}
