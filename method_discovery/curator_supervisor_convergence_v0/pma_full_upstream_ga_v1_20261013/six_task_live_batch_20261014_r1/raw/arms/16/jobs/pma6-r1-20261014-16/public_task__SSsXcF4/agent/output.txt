import pandas as pd
import json

# Read and normalize all sources (reusing previous normalization logic)
with open('/data/source_a/users.json', 'r') as f:
    data_a = json.load(f)
df_a = pd.DataFrame(data_a)

df_b = pd.read_csv('/data/source_b/users.csv')
df_c = pd.read_parquet('/data/source_c/users.parquet')

def normalize_date(date_str):
    if pd.isna(date_str) or date_str is None:
        return None
    try:
        if isinstance(date_str, str):
            if len(date_str) == 10 and date_str[4] == '-' and date_str[7] == '-':
                return date_str
            dt = pd.to_datetime(date_str)
            return dt.strftime('%Y-%m-%d')
        return str(date_str)
    except:
        return str(date_str)

def normalize_status(status_val):
    if pd.isna(status_val):
        return None
    if isinstance(status_val, bool):
        return 'active' if status_val else 'inactive'
    if isinstance(status_val, str):
        if status_val.lower() in ['true', '1', 'yes']:
            return 'active'
        elif status_val.lower() in ['false', '0', 'no']:
            return 'inactive'
        return status_val
    return str(status_val)

# Normalize all sources
df_a_norm = pd.DataFrame({
    'user_id': df_a['id'].astype(int),
    'name': df_a['full_name'],
    'email': df_a['email'],
    'created_date': df_a['registration_date'].apply(normalize_date),
    'status': df_a['status'].apply(normalize_status),
    'source': 'source_a'
})

df_b_norm = pd.DataFrame({
    'user_id': df_b['user_id'].astype(int),
    'name': df_b['name'],
    'email': df_b['email_address'],
    'created_date': df_b['created_at'].apply(normalize_date),
    'status': df_b['is_active'].apply(normalize_status),
    'source': 'source_b'
})

df_c_norm = pd.DataFrame({
    'user_id': df_c['userId'].astype(int),
    'name': df_c['userName'],
    'email': df_c['email'],
    'created_date': df_c['joined'].apply(normalize_date),
    'status': df_c['active'].apply(normalize_status),
    'source': 'source_c'
})

# Combine all sources
all_data = pd.concat([df_a_norm, df_b_norm, df_c_norm], ignore_index=True)

# Define source priority
source_priority = {'source_a': 1, 'source_b': 2, 'source_c': 3}
all_data['priority'] = all_data['source'].map(source_priority)

# Sort by user_id and priority (lower number = higher priority)
all_data = all_data.sort_values(['user_id', 'priority'])

# Track conflicts
conflicts = []
merged_records = []

# Process each unique user
unique_users = all_data['user_id'].unique()

for user_id in unique_users:
    user_records = all_data[all_data['user_id'] == user_id]
    
    if len(user_records) == 1:
        # No conflict, single source
        record = user_records.iloc[0]
        merged_records.append({
            'user_id': int(record['user_id']),
            'name': record['name'],
            'email': record['email'],
            'created_date': record['created_date'],
            'status': record['status']
        })
    else:
        # Multiple sources, check for conflicts
        # Use highest priority source (first in sorted order)
        primary_record = user_records.iloc[0]
        
        merged_record = {
            'user_id': int(primary_record['user_id']),
            'name': primary_record['name'],
            'email': primary_record['email'],
            'created_date': primary_record['created_date'],
            'status': primary_record['status']
        }
        merged_records.append(merged_record)
        
        # Check each field for conflicts
        fields_to_check = ['name', 'email', 'created_date', 'status']
        
        for field in fields_to_check:
            values_by_source = {}
            for _, record in user_records.iterrows():
                source = record['source']
                value = record[field]
                if pd.notna(value):
                    values_by_source[source] = value
            
            # If there are different values across sources, it's a conflict
            unique_values = set(values_by_source.values())
            if len(unique_values) > 1:
                conflict_entry = {
                    'user_id': int(user_id),
                    'field': field,
                    'values': values_by_source,
                    'selected': merged_record[field]
                }
                conflicts.append(conflict_entry)

# Create merged DataFrame
df_merged = pd.DataFrame(merged_records)

# Ensure correct data types
df_merged['user_id'] = df_merged['user_id'].astype(int)

print(\"Merged DataFrame:\")
print(df_merged)
print(f\"\
Total users: {len(df_merged)}\")
print(f\"Total conflicts: {len(conflicts)}\")
print(\"\
Conflicts found:\")
for c in conflicts:
    print(f\"  User {c['user_id']}, field '{c['field']}': {c['values']} -> selected: {c['selected']}\")

# Save merged data to Parquet
df_merged.to_parquet('/app/merged_users.parquet', index=False)
print(\"\
✓ Saved merged_users.parquet\")

# Save conflict report to JSON
conflict_report = {
    'total_conflicts': len(conflicts),
    'conflicts': conflicts
}

with open('/app/conflicts.json', 'w') as f:
    json.dump(conflict_report, f, indent=2)
print(\"✓ Saved conflicts.json\")
",
  "type": "python"
}
