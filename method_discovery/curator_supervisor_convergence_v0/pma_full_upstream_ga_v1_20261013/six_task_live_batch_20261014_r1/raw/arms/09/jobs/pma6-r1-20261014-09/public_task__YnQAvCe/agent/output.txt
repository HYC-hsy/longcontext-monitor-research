def normalize_date(date_value: Any) -> Optional[str]:
    \"\"\"Convert various date formats to YYYY-MM-DD.\"\"\"
    if pd.isna(date_value) or date_value is None or date_value == '':
        return None
    
    date_str = str(date_value).strip()
    
    # Try parsing various formats
    for fmt in ['%Y-%m-%d', '%Y/%m/%d', '%d-%m-%Y', '%d/%m/%Y', '%m-%d-%Y', '%m/%d/%Y']:
        try:
            dt = datetime.strptime(date_str, fmt)
            return dt.strftime('%Y-%m-%d')
        except ValueError:
            continue
    
    # Try pandas datetime parsing
    try:
        dt = pd.to_datetime(date_str)
        return dt.strftime('%Y-%m-%d')
    except:
        pass
    
    return date_str  # Return as-is if can't parse


def map_fields(df: pd.DataFrame, source_name: str) -> pd.DataFrame:
    \"\"\"Map various field names to unified schema.\"\"\"
    mapped_data = {}
    
    for unified_field, possible_names in FIELD_MAPPINGS.items():
        for field_name in possible_names:
            if field_name in df.columns:
                mapped_data[unified_field] = df[field_name]
                break
        else:
            # Field not found, add as None
            mapped_data[unified_field] = None
    
    # Check for status field (optional)
    if 'status' in df.columns:
        mapped_data['status'] = df['status']
    
    # Add source tracking
    result_df = pd.DataFrame(mapped_data)
    result_df['_source'] = source_name
    
    return result_df


def read_source_a(path: str) -> pd.DataFrame:
    \"\"\"Read JSON source (highest priority).\"\"\"
    with open(path, 'r') as f:
        data = json.load(f)
    
    if isinstance(data, dict):
        data = [data]
    
    df = pd.DataFrame(data)
    return map_fields(df, 'source_a')


def read_source_b(path: str) -> pd.DataFrame:
    \"\"\"Read CSV source (medium priority).\"\"\"
    df = pd.read_csv(path)
    return map_fields(df, 'source_b')


def read_source_c(path: str) -> pd.DataFrame:
    \"\"\"Read Parquet source (lowest priority).\"\"\"
    df = pd.read_parquet(path)
    return map_fields(df, 'source_c')


def merge_with_priority(sources: List[pd.DataFrame]) -> tuple[pd.DataFrame, List[Dict]]:
    \"\"\"
    Merge dataframes with priority-based conflict resolution.
    Returns: (merged_df, conflicts_list)
    \"\"\"
    conflicts = []
    
    # Combine all sources
    all_data = pd.concat(sources, ignore_index=True)
    
    # Convert user_id to integer, handling any non-numeric values
    all_data['user_id'] = pd.to_numeric(all_data['user_id'], errors='coerce').astype('Int64')
    
    # Drop rows with invalid user_id
    all_data = all_data.dropna(subset=['user_id'])
    
    # Normalize dates
    all_data['created_date'] = all_data['created_date'].apply(normalize_date)
    
    # Get unique user_ids
    unique_users = all_data['user_id'].unique()
    
    merged_records = []
    
    for user_id in unique_users:
        user_records = all_data[all_data['user_id'] == user_id]
        
        if len(user_records) == 1:
            # No conflict, single source
            record = user_records.iloc[0].to_dict()
            del record['_source']
            merged_records.append(record)
        else:
            # Multiple sources - resolve conflicts
            merged_record = {'user_id': user_id}
            
            # Track which sources have this user
            sources_present = user_records['_source'].tolist()
            
            # For each field, check for conflicts
            for field in ['name', 'email', 'created_date', 'status']:
                field_values = {}
                
                for _, row in user_records.iterrows():
                    source = row['_source']
                    value = row.get(field)
                    
                    # Only track non-null values
                    if pd.notna(value) and value != '' and value is not None:
                        field_values[source] = value
                
                if not field_values:
                    # No values from any source
                    merged_record[field] = None
                elif len(set(field_values.values())) == 1:
                    # All sources agree or only one has a value
                    merged_record[field] = list(field_values.values())[0]
                else:
                    # Conflict detected - use priority
                    selected_value = None
                    for priority_source in SOURCE_PRIORITY:
                        if priority_source in field_values:
                            selected_value = field_values[priority_source]
                            break
                    
                    merged_record[field] = selected_value
                    
                    # Record conflict
                    conflict_entry = {
                        'user_id': int(user_id),
                        'field': field,
                        'values': field_values,
                        'selected': selected_value
                    }
                    conflicts.append(conflict_entry)
            
            merged_records.append(merged_record)
    
    merged_df = pd.DataFrame(merged_records)
    
    # Ensure correct data types
    merged_df['user_id'] = merged_df['user_id'].astype('int64')
    merged_df['name'] = merged_df['name'].astype('string')
    merged_df['email'] = merged_df['email'].astype('string')
    merged_df['created_date'] = merged_df['created_date'].astype('string')
    if 'status' in merged_df.columns:
        merged_df['status'] = merged_df['status'].astype('string')
    
    # Sort by user_id for consistency
    merged_df = merged_df.sort_values('user_id').reset_index(drop=True)
    
    return merged_df, conflicts
