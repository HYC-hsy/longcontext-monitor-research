import pandas as pd
import json
from datetime import datetime
from pathlib import Path
from collections import defaultdict

def normalize_date(date_str):
    \"\"\"Normalize date to YYYY-MM-DD format\"\"\"
    if pd.isna(date_str) or date_str is None:
        return None
    date_str = str(date_str)
    # Already in correct format
    if len(date_str) == 10 and date_str[4] == '-' and date_str[7] == '-':
        return date_str
    # Try parsing
    try:
        dt = pd.to_datetime(date_str)
        return dt.strftime('%Y-%m-%d')
    except:
        return date_str

def normalize_status(status_val):
    \"\"\"Normalize status field to string\"\"\"
    if pd.isna(status_val) or status_val is None:
        return None
    if isinstance(status_val, bool):
        return \"active\" if status_val else \"inactive\"
    if isinstance(status_val, str):
        status_lower = status_val.lower()
        if status_lower in ['true', '1', 'yes', 'active']:
            return \"active\"
        elif status_lower in ['false', '0', 'no', 'inactive']:
            return \"inactive\"
        return status_val
    return str(status_val)

# Read and normalize Source A (JSON - highest priority)
print(\"Reading Source A...\")
with open('/data/source_a/users.json', 'r') as f:
    source_a_raw = json.load(f)

df_a = pd.DataFrame(source_a_raw)
df_a = df_a.rename(columns={
    'id': 'user_id',
    'full_name': 'name',
    'email': 'email',
    'registration_date': 'created_date',
    'status': 'status'
})
df_a['user_id'] = df_a['user_id'].astype(int)
df_a['created_date'] = df_a['created_date'].apply(normalize_date)
df_a['status'] = df_a['status'].apply(normalize_status)
df_a['source'] = 'source_a'
print(f\"Source A: {len(df_a)} records\")

# Read and normalize Source B (CSV - medium priority)
print(\"Reading Source B...\")
df_b = pd.read_csv('/data/source_b/users.csv')
df_b = df_b.rename(columns={
    'user_id': 'user_id',
    'name': 'name',
    'email_address': 'email',
    'created_at': 'created_date',
    'is_active': 'status'
})
df_b['user_id'] = df_b['user_id'].astype(int)
df_b['created_date'] = df_b['created_date'].apply(normalize_date)
df_b['status'] = df_b['status'].apply(normalize_status)
df_b['source'] = 'source_b'
print(f\"Source B: {len(df_b)} records\")

# Read and normalize Source C (Parquet - lowest priority)
print(\"Reading Source C...\")
df_c = pd.read_parquet('/data/source_c/users.parquet')
df_c = df_c.rename(columns={
    'userId': 'user_id',
    'userName': 'name',
    'email': 'email',
    'joined': 'created_date',
    'active': 'status'
})
df_c['user_id'] = df_c['user_id'].astype(int)
df_c['created_date'] = df_c['created_date'].apply(normalize_date)
df_c['status'] = df_c['status'].apply(normalize_status)
df_c['source'] = 'source_c'
print(f\"Source C: {len(df_c)} records\")

# Combine all sources
all_data = pd.concat([df_a, df_b, df_c], ignore_index=True)
print(f\"\
Total records: {len(all_data)}\")
print(f\"Unique users: {all_data['user_id'].nunique()}\")

# Track conflicts
conflicts = []
merged_records = {}

# Get all unique user IDs
all_user_ids = sorted(all_data['user_id'].unique())

for user_id in all_user_ids:
    user_records = all_data[all_data['user_id'] == user_id]
    
    # If only one source, no conflict
    if len(user_records) == 1:
        record = user_records.iloc[0]
        merged_records[user_id] = {
            'user_id': int(user_id),
            'name': record['name'],
            'email': record['email'],
            'created_date': record['created_date'],
            'status': record['status']
        }
    else:
        # Multiple sources - need to merge with priority
        sources_present = user_records['source'].tolist()
        
        # Priority order: source_a > source_b > source_c
        priority_order = ['source_a', 'source_b', 'source_c']
        
        merged_record = {'user_id': int(user_id)}
        
        # For each field, check for conflicts
        for field in ['name', 'email', 'created_date', 'status']:
            values_by_source = {}
            for _, row in user_records.iterrows():
                source = row['source']
                value = row[field]
                if pd.notna(value):
                    values_by_source[source] = value
            
            # Select value based on priority
            selected_value = None
            for source in priority_order:
                if source in values_by_source:
                    selected_value = values_by_source[source]
                    break
            
            merged_record[field] = selected_value
            
            # Check if there's a conflict (different values across sources)
            unique_values = set(v for v in values_by_source.values() if v is not None)
            if len(unique_values) > 1:
                conflict_entry = {
                    'user_id': int(user_id),
                    'field': field,
                    'values': {},
                    'selected': selected_value
                }
                
                # Add values from each source
                for source in ['source_a', 'source_b', 'source_c']:
                    if source in values_by_source:
                        conflict_entry['values'][source] = values_by_source[source]
                
                conflicts.append(conflict_entry)
        
        merged_records[user_id] = merged_record

# Create merged dataframe
merged_df = pd.DataFrame(list(merged_records.values()))

# Ensure correct column order and types
merged_df = merged_df[['user_id', 'name', 'email', 'created_date', 'status']]
merged_df['user_id'] = merged_df['user_id'].astype(int)

print(f\"\
Merged dataset: {len(merged_df)} unique users\")
print(f\"Conflicts found: {len(conflicts)}\")
print(\"\
Merged data preview:\")
print(merged_df)

# Save merged data to parquet
output_parquet = '/app/merged_users.parquet'
merged_df.to_parquet(output_parquet, index=False)
print(f\"\
✓ Saved merged data to {output_parquet}\")

# Save conflict report
conflict_report = {
    'total_conflicts': len(conflicts),
    'conflicts': conflicts
}

output_json = '/app/conflicts.json'
with open(output_json, 'w') as f:
    json.dump(conflict_report, f, indent=2)
print(f\"✓ Saved conflict report to {output_json}\")

print(\"\
=== Summary ===\")
print(f\"Total unique users: {len(merged_df)}\")
print(f\"Total conflicts: {len(conflicts)}\")
print(f\"Users from source_a: {len(df_a)}\")
print(f\"Users from source_b: {len(df_b)}\")
print(f\"Users from source_c: {len(df_c)}\")
"
}
