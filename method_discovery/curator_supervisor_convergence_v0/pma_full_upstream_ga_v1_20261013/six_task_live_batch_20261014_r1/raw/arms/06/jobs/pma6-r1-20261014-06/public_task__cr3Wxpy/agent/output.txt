\"\"\"
Python reimplementation of COBOL BOOKFORUM program.
Reads INPUT.DAT and processes book transactions, updating ACCOUNTS.DAT, 
BOOKS.DAT, and TRANSACTIONS.DAT files.
\"\"\"

import sys
import os


def read_input_file(filepath):
    \"\"\"Read and parse the input file.\"\"\"
    try:
        with open(filepath, 'rb') as f:
            data = f.read()
        
        # Parse fixed-width input record
        # Based on actual file: U001U002B007200 (15 bytes)
        buyer_id = data[0:4].decode('ascii')
        seller_id = data[4:8].decode('ascii')
        book_id = data[8:12].decode('ascii')
        # Amount is the remaining bytes, parse as integer
        amount_str = data[12:].decode('ascii')
        amount = int(amount_str)
        
        return buyer_id, seller_id, book_id, amount
    except FileNotFoundError:
        print(\"Error: Input file is empty\")
        sys.exit(1)
    except Exception as e:
        print(f\"Error reading input file: {e}\")
        sys.exit(1)


def read_accounts_file(filepath):
    \"\"\"Read accounts file and return list of account records.\"\"\"
    accounts = []
    with open(filepath, 'rb') as f:
        data = f.read()
    
    # Each account record is 34 bytes:
    # ACCOUNT-ID (4) + ACCOUNT-NAME (20) + ACCOUNT-BALANCE (10)
    record_size = 34
    for i in range(0, len(data), record_size):
        if i + record_size > len(data):
            break
        record = data[i:i+record_size]
        account_id = record[0:4].decode('ascii')
        account_name = record[4:24].decode('ascii')
        account_balance = int(record[24:34].decode('ascii'))
        accounts.append({
            'id': account_id,
            'name': account_name,
            'balance': account_balance
        })
    
    return accounts


def write_accounts_file(filepath, accounts):
    \"\"\"Write accounts back to file in fixed-width format.\"\"\"
    with open(filepath, 'wb') as f:
        for account in accounts:
            # Format: ID(4) + NAME(20) + BALANCE(10 digits, zero-padded)
            record = (
                account['id'].ljust(4)[:4] +
                account['name'].ljust(20)[:20] +
                f\"{account['balance']:010d}\"
            )
            f.write(record.encode('ascii'))


def read_books_file(filepath):
    \"\"\"Read books file and return list of book records.\"\"\"
    books = []
    with open(filepath, 'rb') as f:
        data = f.read()
    
    # Each book record is 28 bytes:
    # BOOK-ID (4) + BOOK-TITLE (20) + BOOK-OWNER (4)
    record_size = 28
    for i in range(0, len(data), record_size):
        if i + record_size > len(data):
            break
        record = data[i:i+record_size]
        book_id = record[0:4].decode('ascii')
        book_title = record[4:24].decode('ascii')
        book_owner = record[24:28].decode('ascii')
        books.append({
            'id': book_id,
            'title': book_title,
            'owner': book_owner
        })
    
    return books


def write_books_file(filepath, books):
    \"\"\"Write books back to file in fixed-width format.\"\"\"
    with open(filepath, 'wb') as f:
        for book in books:
            # Format: ID(4) + TITLE(20) + OWNER(4)
            record = (
                book['id'].ljust(4)[:4] +
                book['title'].ljust(20)[:20] +
                book['owner'].ljust(4)[:4]
            )
            f.write(record.encode('ascii'))


def append_transaction_file(filepath, trans_book, trans_amount, trans_seller, trans_buyer):
    \"\"\"Append a transaction record to the transactions file.\"\"\"
    with open(filepath, 'ab') as f:
        # Format: TRANS-BOOK(4) + TRANS-AMOUNT(10) + TRANS-SELLER(4) + TRANS-BUYER(4)
        record = (
            trans_book.ljust(4)[:4] +
            f\"{trans_amount:010d}\" +
            trans_seller.ljust(4)[:4] +
            trans_buyer.ljust(4)[:4]
        )
        f.write(record.encode('ascii'))


def main():
    # File paths
    input_file = 'src/INPUT.DAT'
    accounts_file = 'data/ACCOUNTS.DAT'
    books_file = 'data/BOOKS.DAT'
    transactions_file = 'data/TRANSACTIONS.DAT'
    
    # Read input
    buyer_id, seller_id, book_id, amount = read_input_file(input_file)
    
    # Initialize validation flags
    buyer_found = False
    seller_found = False
    book_found = False
    valid_owner = False
    
    # Validate users in accounts file
    accounts = read_accounts_file(accounts_file)
    for account in accounts:
        if account['id'] == buyer_id:
            buyer_found = True
        if account['id'] == seller_id:
            seller_found = True
    
    # Validate book and ownership
    books = read_books_file(books_file)
    for book in books:
        if book['id'] == book_id:
            book_found = True
            if book['owner'] == seller_id:
                valid_owner = True
    
    # Display blank line (from COBOL line 119)
    print(\" \")
    
    # Process transaction if all validations pass
    if buyer_found and seller_found and book_found and valid_owner:
        # Update account balances
        for account in accounts:
            if account['id'] == buyer_id:
                account['balance'] -= amount
            if account['id'] == seller_id:
                account['balance'] += amount
        
        # Write updated accounts
        write_accounts_file(accounts_file, accounts)
        
        # Update book ownership
        for book in books:
            if book['id'] == book_id:
                book['owner'] = buyer_id
        
        # Write updated books
        write_books_file(books_file, books)
        
        # Append transaction record
        append_transaction_file(transactions_file, book_id, amount, seller_id, buyer_id)
        
        print(\"Transaction completed successfully\")
    else:
        print(\"Transaction failed due to validation errors\")


if __name__ == \"__main__\":
    main()
"
}
