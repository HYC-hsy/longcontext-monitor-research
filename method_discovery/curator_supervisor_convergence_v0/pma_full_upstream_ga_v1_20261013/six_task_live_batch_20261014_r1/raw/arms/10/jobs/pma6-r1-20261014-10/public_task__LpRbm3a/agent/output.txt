    elif col in ['joined', 'created_at', 'registration_date']:
        source_c_mapping[col] = 'created_date'
    elif col == 'status':

field_mappings['source_c'] = source_c_mapping

print(f\"=== Field Mappings ===\")

print(f\"Source B: {field_mappings['source_b']}\")
print(f\"Source C: {field_mappings['source_c']}\
\")

# Normalize date format function
def normalize_date(date_val):
    \"\"\"Convert date to YYYY-MM-DD format\"\"\"
    if pd.isna(date_val) or date_val is None:
        return None
    
    date_str = str(date_val)
    
    # Try various date formats
    for fmt in ['%Y-%m-%d', '%Y/%m/%d', '%m/%d/%Y', '%d/%m/%Y', '%Y-%m-%d %H:%M:%S']:
        try:
            dt = datetime.strptime(date_str.split()[0], fmt)
            return dt.strftime('%Y-%m-%d')
        except:
            continue
    
    # If already in YYYY-MM-DD format
    if len(date_str) >= 10:
        return date_str[:10]
    
    return date_str

# Apply field mappings and normalize
def normalize_dataframe(df, mappings, source_name):
    \"\"\"Normalize a dataframe with field mappings\"\"\"
    df_norm = pd.DataFrame()
    
    for old_col, new_col in mappings.items():
        if old_col in df.columns:
            df_norm[new_col] = df[old_col]
    
    # Normalize dates
    if 'created_date' in df_norm.columns:
        df_norm['created_date'] = df_norm['created_date'].apply(normalize_date)
    
    # Ensure user_id is integer
    if 'user_id' in df_norm.columns:
        df_norm['user_id'] = pd.to_numeric(df_norm['user_id'], errors='coerce').astype('Int64')
    
    # Add source column for tracking
    df_norm['_source'] = source_name
    
    return df_norm

# Normalize all sources
df_a_norm = normalize_dataframe(df_a, field_mappings['source_a'], 'source_a')
df_b_norm = normalize_dataframe(df_b, field_mappings['source_b'], 'source_b')
df_c_norm = normalize_dataframe(df_c, field_mappings['source_c'], 'source_c')

print(\"=== Normalized DataFrames ===\")
print(\"Source A normalized:\")
print(df_a_norm)
print(\"\
Source B normalized:\")
print(df_b_norm)
print(\"\
Source C normalized:\")
print(df_c_norm)
print()

# Merge with priority handling and conflict detection
conflicts = []
merged_records = {}

# Process sources in priority order: A > B > C
sources = [
    ('source_a', df_a_norm),
    ('source_b', df_b_norm),
    ('source_c', df_c_norm)
]

for source_name, df_source in sources:
    for _, row in df_source.iterrows():
        user_id = row['user_id']
        
        if pd.isna(user_id):
            continue
        
        user_id = int(user_id)
        
        if user_id not in merged_records:
            # First time seeing this user
            merged_records[user_id] = {
                'user_id': user_id,
                'name': row.get('name'),
                'email': row.get('email'),
                'created_date': row.get('created_date'),
                'status': row.get('status'),
                '_sources': {source_name: row.to_dict()}
            }
        else:
            # User exists - check for conflicts
            existing = merged_records[user_id]
            existing['_sources'][source_name] = row.to_dict()
            
            # Check each field for conflicts
            for field in ['name', 'email', 'created_date', 'status']:
                new_val = row.get(field)
                existing_val = existing[field]
                
                # Only consider it a conflict if both values exist and differ
                if pd.notna(new_val) and pd.notna(existing_val) and str(new_val) != str(existing_val):
                    # Record conflict
                    conflict_entry = {
                        'user_id': user_id,
                        'field': field,
                        'values': {},
                        'selected': str(existing_val)  # Existing value wins (higher priority)
                    }
                    
                    # Collect all values from all sources
                    for src_name, src_data in existing['_sources'].items():
                        val = src_data.get(field)
                        if pd.notna(val):
                            conflict_entry['values'][src_name] = str(val)
                    
                    conflicts.append(conflict_entry)
                
                # Fill in missing values from lower priority sources
                if pd.isna(existing_val) and pd.notna(new_val):
                    existing[field] = new_val

print(f\"=== Merge Summary ===\")
print(f\"Total unique users: {len(merged_records)}\")
print(f\"Total conflicts detected: {len(conflicts)}\
\")

# Create final merged dataframe
merged_data = []
for user_id, record in merged_records.items():
    merged_data.append({
        'user_id': record['user_id'],
        'name': record['name'],
        'email': record['email'],
        'created_date': record['created_date'],
        'status': record['status']
    })

df_merged = pd.DataFrame(merged_data)

# Ensure correct data types
df_merged['user_id'] = df_merged['user_id'].astype('int64')

# Sort by user_id for consistency
df_merged = df_merged.sort_values('user_id').reset_index(drop=True)

print(\"=== Merged Data ===\")
print(df_merged)
print(f\"\
Data types:\
{df_merged.dtypes}\
\")

# Save merged data to parquet
output_path = '/app/merged_users.parquet'
df_merged.to_parquet(output_path, index=False)
print(f\"✓ Saved merged data to {output_path}\")

# Create conflict report
conflict_report = {
    'total_conflicts': len(conflicts),
    'conflicts': conflicts
}

# Save conflict report
conflict_path = '/app/conflicts.json'
with open(conflict_path, 'w') as f:
    json.dump(conflict_report, f, indent=2)
print(f\"✓ Saved conflict report to {conflict_path}\")

print(f\"\
=== Success ===\")
print(f\"✓ Merged {len(merged_records)} unique users\")
print(f\"✓ Detected {len(conflicts)} conflicts\")
print(f\"✓ Output files created successfully\")
