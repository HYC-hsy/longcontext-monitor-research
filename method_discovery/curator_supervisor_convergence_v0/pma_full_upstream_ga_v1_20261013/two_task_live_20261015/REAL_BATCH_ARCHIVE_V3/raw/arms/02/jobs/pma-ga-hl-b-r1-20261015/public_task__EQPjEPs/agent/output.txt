import time
from base_terminal import BaseTerminal


class HeadlessTerminal(BaseTerminal):
    \"\"\"
    A headless terminal implementation that provides an interactive bash shell
    with full terminal emulation including support for interactive programs
    and modifier keys.
    \"\"\"
    
    def __init__(self):
        \"\"\"
        Initialize a headless terminal with an interactive bash shell.
        
        The shell is started in interactive mode which sources ~/.bashrc
        and provides full terminal functionality through a pseudo-terminal (pty).
        \"\"\"
        # Start bash in interactive mode (-i) which will source ~/.bashrc
        # The pty provides full terminal emulation for interactive programs
        self.process = pexpect.spawn(
            '/bin/bash',
            ['-i'],
            encoding='utf-8',
            echo=False,  # Avoid double-echoing
            timeout=None  # No timeout on operations
        )
        
        # Give the shell time to initialize and source startup files
        time.sleep(0.2)
    
    def send_keystrokes(self, keystrokes: str, wait_sec: float = 0.0) -> None:
        \"\"\"
        Sends keystrokes to the terminal.
        
        Supports all standard characters as well as control sequences like:
        - \\\\x03 for Ctrl+C (SIGINT)
        - \\\\x04 for Ctrl+D (EOF)
        - \\\\x1a for Ctrl+Z (SIGTSTP)
        - \\\\r or \\\
 for Enter
        
        Args:
            keystrokes: The keystrokes to send to the terminal.
                       Can include special characters and control sequences.
            wait_sec: The number of seconds to wait after sending keystrokes
                     for the command to complete.
        \"\"\"
        self.process.send(keystrokes)
        
        if wait_sec > 0:
            time.sleep(wait_sec)
    
    def read_output(self, timeout: float = 0.1) -> str:
        \"\"\"
        Read available output from the terminal.
        
        Args:
            timeout: Maximum time to wait for output in seconds.
        
        Returns:
            The output read from the terminal.
        \"\"\"
        try:
            # Try to read with a short timeout
            output = self.process.read_nonblocking(size=8192, timeout=timeout)
            return output
        except pexpect.TIMEOUT:
            return \"\"
        except pexpect.EOF:
            return \"\"
    
    def is_alive(self) -> bool:
        \"\"\"Check if the terminal process is still running.\"\"\"
        return self.process.isalive()
    
    def close(self):
        \"\"\"Close the terminal and terminate the bash process.\"\"\"
        if hasattr(self, 'process') and self.process.isalive():
            self.process.terminate(force=True)
    
    def __del__(self):
        \"\"\"Cleanup: terminate the bash process when the object is destroyed.\"\"\"
        self.close()
    
    def __enter__(self):
        \"\"\"Context manager entry.\"\"\"
        return self
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        \"\"\"Context manager exit.\"\"\"
        self.close()
        return False
"

        \"\"\"
        Read available output from the terminal.
        
        Args:
            timeout: Maximum time to wait for output in seconds.
        
        Returns:
            The output read from the terminal.
        \"\"\"
        output_parts = []
        start_time = time.time()
        
        while True:
            try:
                remaining_time = timeout - (time.time() - start_time)
                if remaining_time <= 0:
                    break
                    
                # Read available data with a short timeout
                chunk = self.process.read_nonblocking(size=8192, timeout=min(remaining_time, 0.1))
                if chunk:
                    output_parts.append(chunk)
            except pexpect.TIMEOUT:
                # No more data available, break out
                break
            except pexpect.EOF:
                # Process has ended
                break
        
        return ''.join(output_parts)"
