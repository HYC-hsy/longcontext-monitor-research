  "content": "import pexpect
import time
from base_terminal import BaseTerminal


class HeadlessTerminal(BaseTerminal):
    \"\"\"
    A headless terminal implementation that provides an interactive bash shell
    with full terminal emulation support using pexpect.
    \"\"\"
    
    def __init__(self, timeout=30):
        \"\"\"
        Initialize the headless terminal with an interactive bash shell.
        
        Args:
            timeout: Default timeout for terminal operations in seconds
        \"\"\"
        self.timeout = timeout
        
        # Spawn an interactive bash shell that sources startup files
        # -i: interactive mode (sources ~/.bashrc)
        # The shell will run in a pseudo-terminal (PTY) which provides full terminal emulation
        self.process = pexpect.spawn(
            '/bin/bash',
            args=['--login', '-i'],  # --login sources profile files, -i sources bashrc
            encoding='utf-8',
            echo=False,  # Disable echo to avoid duplicate output
            timeout=self.timeout
        )
        
        # Set a reasonable window size for the terminal
        self.process.setwinsize(24, 80)
        
        # Wait a moment for the shell to initialize
        time.sleep(0.1)
    
    def send_keystrokes(self, keystrokes: str, wait_sec: float = 0.0) -> None:
        \"\"\"
        Sends keystrokes to the terminal.
        
        This method supports:
        - Regular text input
        - Control characters (e.g., \\x03 for Ctrl+C)
        - Special keys
        - Interactive programs
        
        Args:
            keystrokes: The keystrokes to send to the terminal.
                       Can include control characters like \\x03 (Ctrl+C),
                       \\r or \
 for Enter, etc.
            wait_sec: The number of seconds to wait after sending the keystrokes.
        \"\"\"
        # Send the keystrokes to the process
        self.process.send(keystrokes)
        
        # Wait if requested
        if wait_sec > 0:
            time.sleep(wait_sec)
    
    def read_output(self, timeout=1.0):
        \"\"\"
        Read available output from the terminal.
        
        Args:
            timeout: How long to wait for output in seconds
            
        Returns:
            String containing the output
        \"\"\"
        try:
            # Read whatever is available before the timeout
            output = self.process.read_nonblocking(size=8192, timeout=timeout)
            return output
        except pexpect.TIMEOUT:
            return \"\"
        except pexpect.EOF:
            return \"\"
    
    def get_buffer(self):
        \"\"\"
        Get the current terminal buffer content.
        
        Returns:
            String containing the buffer
        \"\"\"
        return self.process.before if self.process.before else \"\"
    
    def is_alive(self):
        \"\"\"
        Check if the terminal process is still running.
        
        Returns:
            True if the process is alive, False otherwise
        \"\"\"
        return self.process.isalive()
    
    def close(self):
        \"\"\"
        Close the terminal session.
        \"\"\"
        if self.is_alive():
            # Try to exit gracefully first
            try:
                self.send_keystrokes('exit\
')
                self.process.expect(pexpect.EOF, timeout=2)
            except (pexpect.TIMEOUT, pexpect.EOF):
                pass
            
            # Force termination if still alive
            if self.is_alive():
                self.process.terminate(force=True)
    
    def __enter__(self):
        \"\"\"Context manager entry.\"\"\"
        return self
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        \"\"\"Context manager exit.\"\"\"
        self.close()
        return False
    
    def __del__(self):
        \"\"\"Cleanup on deletion.\"\"\"
        try:
            self.close()
        except:
            pass
