import random
import string


class GrammarFuzzer:
    def __init__(self, target_name):
        self.target_name = target_name
        self.identifiers = ['id', 'name', 'value', 'user', 'age', 'email', 'count', 'price', 
                           'users', 'products', 'orders', 'table1', 'table2', 't1', 't2']
        self.strings = [\"'test'\", \"'hello'\", \"'world'\", \"''\", \"'a'\", 
                       '\"test\"', '\"hello\"', '\"\"', \"'with space'\"]
        self.numbers = ['0', '1', '42', '100', '3.14', '0.5', '1e10', '2.5e-3']
        
    def random_identifier(self):
        return random.choice(self.identifiers)
    
    def random_string(self):
        return random.choice(self.strings)
    
    def random_number(self):
        return random.choice(self.numbers)
    
    def random_value(self):
        choice = random.randint(0, 10)
        if choice < 4:
            return self.random_number()
        elif choice < 7:
            return self.random_string()
        elif choice < 8:
            return 'NULL'
        elif choice < 9:
            return 'TRUE'
        else:
            return 'FALSE'


class SQLFuzzer(GrammarFuzzer):
    def generate_expression(self, depth=0):
        \"\"\"Generate SQL expressions with various complexity levels\"\"\"
        if depth > 3:
            return self.random_value()
        
        choice = random.randint(0, 20)
        
        # Simple values
        if choice < 5:
            return self.random_value()
        
        # Identifier or qualified identifier
        elif choice < 8:
            if random.random() < 0.3:
                return f\"{self.random_identifier()}.{self.random_identifier()}\"
            return self.random_identifier()
        
        # Binary operations
        elif choice < 11:
            ops = ['+', '-', '*', '/', '%', '||', '=', '<', '>', '<=', '>=', '<>', '!=']
            left = self.generate_expression(depth + 1)
            right = self.generate_expression(depth + 1)
            return f\"{left} {random.choice(ops)} {right}\"
        
        # Logical operations
        elif choice < 13:
            left = self.generate_expression(depth + 1)
            right = self.generate_expression(depth + 1)
            op = random.choice(['AND', 'OR'])
            return f\"{left} {op} {right}\"
        
        # NOT expression
        elif choice == 13:
            return f\"NOT {self.generate_expression(depth + 1)}\"
        
        # Parenthesized expression
        elif choice == 14:
            return f\"({self.generate_expression(depth + 1)})\"
        
        # IS NULL / IS NOT NULL
        elif choice == 15:
            expr = self.generate_expression(depth + 1)
            return f\"{expr} IS {'NOT ' if random.random() < 0.5 else ''}NULL\"
        
        # BETWEEN
        elif choice == 16:
            expr = self.generate_expression(depth + 1)
            low = self.random_value()
            high = self.random_value()
            not_clause = 'NOT ' if random.random() < 0.3 else ''
            return f\"{expr} {not_clause}BETWEEN {low} AND {high}\"
        
        # IN clause
        elif choice == 17:
            expr = self.generate_expression(depth + 1)
            values = ', '.join([self.random_value() for _ in range(random.randint(1, 4))])
            not_clause = 'NOT ' if random.random() < 0.3 else ''
            return f\"{expr} {not_clause}IN ({values})\"
        
        # LIKE
        elif choice == 18:
            expr = self.generate_expression(depth + 1)
            pattern = random.choice([\"'%test%'\", \"'test%'\", \"'%test'\", \"'_test'\"])
            not_clause = 'NOT ' if random.random() < 0.3 else ''
            return f\"{expr} {not_clause}LIKE {pattern}\"
        
        # Function call
        elif choice == 19:
            func = random.choice(['COUNT', 'SUM', 'AVG', 'MAX', 'MIN', 'UPPER', 'LOWER'])
            if func == 'COUNT' and random.random() < 0.5:
                return f\"{func}(*)\"
            args = ', '.join([self.generate_expression(depth + 1) for _ in range(random.randint(0, 2))])
            return f\"{func}({args})\"
        
        # CASE expression
        else:
            case_expr = f\"CASE\"
            if random.random() < 0.3:
                case_expr += f\" {self.generate_expression(depth + 1)}\"
            for _ in range(random.randint(1, 2)):
                case_expr += f\" WHEN {self.generate_expression(depth + 1)} THEN {self.generate_expression(depth + 1)}\"
            if random.random() < 0.5:
                case_expr += f\" ELSE {self.generate_expression(depth + 1)}\"
            case_expr += \" END\"
            return case_expr
    
    def generate_select(self):
        \"\"\"Generate SELECT statements with various clauses\"\"\"
        stmt = \"SELECT\"
        
        # DISTINCT/ALL
        if random.random() < 0.2:
            stmt += \" DISTINCT\"
        elif random.random() < 0.1:
            stmt += \" ALL\"
        
        # Select list
        if random.random() < 0.2:
            stmt += \" *\"
        else:
            cols = []
            for _ in range(random.randint(1, 4)):
                expr = self.generate_expression()
                if random.random() < 0.4:
                    alias = self.random_identifier()
                    if random.random() < 0.5:
                        expr += f\" AS {alias}\"
                    else:
                        expr += f\" {alias}\"
                cols.append(expr)
            stmt += \" \" + \", \".join(cols)
        
        # FROM clause
        if random.random() < 0.9:
            stmt += \" FROM\"
            
            # Table reference or subquery
            if random.random() < 0.8:
                table = self.random_identifier()
                if random.random() < 0.3:
                    alias = self.random_identifier()
                    if random.random() < 0.5:
                        table += f\" AS {alias}\"
                    else:
                        table += f\" {alias}\"
                stmt += f\" {table}\"
            else:
                # Subquery
                subquery = self.generate_select()
                alias = self.random_identifier()
                if random.random() < 0.5:
                    stmt += f\" ({subquery}) AS {alias}\"
                else:
                    stmt += f\" ({subquery}) {alias}\"
            
            # JOIN clauses
            if random.random() < 0.4:
                for _ in range(random.randint(1, 2)):
                    join_type = random.choice(['', 'INNER ', 'LEFT ', 'RIGHT ', 'FULL ', 
                                              'LEFT OUTER ', 'RIGHT OUTER ', 'FULL OUTER ', 'CROSS '])
                    stmt += f\" {join_type}JOIN {self.random_identifier()}\"
                    if join_type != 'CROSS ' and random.random() < 0.8:
                        stmt += f\" ON {self.generate_expression()}\"
        
        # WHERE clause
        if random.random() < 0.6:
            stmt += f\" WHERE {self.generate_expression()}\"
        
        # GROUP BY clause
        if random.random() < 0.3:
            stmt += \" GROUP BY \" + \", \".join([self.generate_expression() for _ in range(random.randint(1, 2))])
            
            # HAVING clause
            if random.random() < 0.5:
                stmt += f\" HAVING {self.generate_expression()}\"
        
        # ORDER BY clause
        if random.random() < 0.4:
            orders = []
            for _ in range(random.randint(1, 2)):
                expr = self.generate_expression()
                direction = random.choice(['', ' ASC', ' DESC'])
                orders.append(expr + direction)
            stmt += \" ORDER BY \" + \", \".join(orders)
        
        # LIMIT clause
        if random.random() < 0.3:
            stmt += f\" LIMIT {random.randint(1, 100)}\"
            if random.random() < 0.4:
                stmt += f\" OFFSET {random.randint(0, 50)}\"
        
        # UNION/EXCEPT/INTERSECT
        if random.random() < 0.2:
            op = random.choice(['UNION', 'EXCEPT', 'INTERSECT'])
            stmt += f\" {op}\"
            if random.random() < 0.3:
                stmt += \" ALL\"
            stmt += f\" {self.generate_select()}\"
        
        return stmt
    
    def generate_insert(self):
        \"\"\"Generate INSERT statements\"\"\"
        stmt = f\"INSERT INTO {self.random_identifier()}\"
        
        # Column list
        if random.random() < 0.6:
            cols = [self.random_identifier() for _ in range(random.randint(1, 4))]
            stmt += f\" ({', '.join(cols)})\"
        
        # VALUES or SELECT
        if random.random() < 0.8:
            stmt += \" VALUES\"
            for i in range(random.randint(1, 3)):
                if i > 0:
                    stmt += \",\"
                values = [self.generate_expression() for _ in range(random.randint(1, 4))]
                stmt += f\" ({', '.join(values)})\"
        else:
            stmt += f\" {self.generate_select()}\"
        
        return stmt
    
    def generate_update(self):
        \"\"\"Generate UPDATE statements\"\"\"
        stmt = f\"UPDATE {self.random_identifier()} SET\"
        
        assignments = []
        for _ in range(random.randint(1, 4)):
            assignments.append(f\"{self.random_identifier()} = {self.generate_expression()}\")
        stmt += \" \" + \", \".join(assignments)
        
        if random.random() < 0.7:
            stmt += f\" WHERE {self.generate_expression()}\"
        
        return stmt
    
    def generate_delete(self):
        \"\"\"Generate DELETE statements\"\"\"
        stmt = f\"DELETE FROM {self.random_identifier()}\"
        
        if random.random() < 0.7:
            stmt += f\" WHERE {self.generate_expression()}\"
        
        return stmt
    
    def generate_create_table(self):
        \"\"\"Generate CREATE TABLE statements\"\"\"
        stmt = \"CREATE TABLE\"
        
        if random.random() < 0.3:
            stmt += \" IF NOT EXISTS\"
        
        stmt += f\" {self.random_identifier()} (\"
        
        columns = []
        for _ in range(random.randint(1, 5)):
            col = self.random_identifier()
            dtype = random.choice(['INT', 'INTEGER', 'VARCHAR', 'TEXT', 'FLOAT', 
                                  'DOUBLE', 'BOOLEAN', 'DATE', 'TIMESTAMP'])
            col += f\" {dtype}\"
            
            # Column constraints
            if random.random() < 0.2:
                col += \" PRIMARY KEY\"
            if random.random() < 0.2:
                col += \" UNIQUE\"
            if random.random() < 0.2:
                col += \" NOT NULL\"
            if random.random() < 0.1:
                col += f\" DEFAULT {self.random_value()}\"
            if random.random() < 0.1:
                ref_table = self.random_identifier()
                ref_col = self.random_identifier()
                col += f\" REFERENCES {ref_table}({ref_col})\"
                if random.random() < 0.5:
                    col += \" ON DELETE \" + random.choice(['CASCADE', 'SET NULL'])
            
            columns.append(col)
        
        stmt += \", \".join(columns)
        
        # Table constraints
        if random.random() < 0.2:
            constraint_type = random.choice(['PRIMARY', 'UNIQUE', 'FOREIGN'])
            if constraint_type == 'PRIMARY':
                cols = [self.random_identifier() for _ in range(random.randint(1, 2))]
                stmt += f\", PRIMARY KEY ({', '.join(cols)})\"
            elif constraint_type == 'UNIQUE':
                cols = [self.random_identifier() for _ in range(random.randint(1, 2))]
                stmt += f\", UNIQUE ({', '.join(cols)})\"
            else:
                cols = [self.random_identifier() for _ in range(random.randint(1, 2))]
                ref_table = self.random_identifier()
                ref_cols = [self.random_identifier() for _ in range(len(cols))]
                stmt += f\", FOREIGN KEY ({', '.join(cols)}) REFERENCES {ref_table} ({', '.join(ref_cols)})\"
        
        stmt += \")\"
        return stmt
    
    def generate_drop_table(self):
        \"\"\"Generate DROP TABLE statements\"\"\"
        stmt = \"DROP TABLE\"
        
        if random.random() < 0.5:
            stmt += \" IF EXISTS\"
        
        stmt += f\" {self.random_identifier()}\"
        return stmt
    
    def generate_alter_table(self):
        \"\"\"Generate ALTER TABLE statements\"\"\"
        stmt = f\"ALTER TABLE {self.random_identifier()}\"
        
        action = random.choice(['ADD', 'DROP', 'MODIFY', 'RENAME'])
        
        if action == 'ADD':
            if random.random() < 0.5:
                stmt += \" ADD COLUMN\"
            else:
                stmt += \" ADD\"
            col = self.random_identifier()
            dtype = random.choice(['INT', 'VARCHAR', 'TEXT', 'FLOAT'])
            stmt += f\" {col} {dtype}\"
        elif action == 'DROP':
            if random.random() < 0.5:
                stmt += \" DROP COLUMN\"
            else:
                stmt += \" DROP\"
            stmt += f\" {self.random_identifier()}\"
        elif action == 'MODIFY':
            if random.random() < 0.5:
                stmt += \" MODIFY COLUMN\"
            else:
                stmt += \" MODIFY\"
            col = self.random_identifier()
            dtype = random.choice(['INT', 'VARCHAR', 'TEXT', 'FLOAT'])
            stmt += f\" {col} {dtype}\"
        else:  # RENAME
            stmt += f\" RENAME TO {self.random_identifier()}\"
        
        return stmt
    
    def generate_valid_statement(self):
        \"\"\"Generate a valid SQL statement\"\"\"
        stmt_type = random.choice(['SELECT', 'SELECT', 'SELECT', 'INSERT', 'UPDATE', 
                                  'DELETE', 'CREATE', 'DROP', 'ALTER'])
        
        if stmt_type == 'SELECT':
            return self.generate_select()
        elif stmt_type == 'INSERT':
            return self.generate_insert()
        elif stmt_type == 'UPDATE':
            return self.generate_update()
        elif stmt_type == 'DELETE':
            return self.generate_delete()
        elif stmt_type == 'CREATE':
            return self.generate_create_table()
        elif stmt_type == 'DROP':
            return self.generate_drop_table()
        else:  # ALTER
            return self.generate_alter_table()
    
    def generate_invalid_input(self):
        \"\"\"Generate invalid inputs to test error handling\"\"\"
        choice = random.randint(0, 10)
        
        if choice < 2:
            # Empty or whitespace
            return random.choice(['', ' ', '  ', '\
', '\\t'])
        elif choice < 4:
            # Random keywords without structure
            keywords = ['SELECT', 'FROM', 'WHERE', 'INSERT', 'UPDATE', 'DELETE', 
                       'CREATE', 'TABLE', 'DROP', 'ALTER']
            return ' '.join(random.choices(keywords, k=random.randint(1, 5)))
        elif choice < 6:
            # Incomplete statements
            return random.choice([
                'SELECT',
                'SELECT * FROM',
                'INSERT INTO',
                'UPDATE SET',
                'DELETE FROM',
                'CREATE TABLE',
                'DROP TABLE',
                'ALTER TABLE',
                'SELECT * FROM WHERE',
            ])
        elif choice < 8:
            # Random punctuation
            return ''.join(random.choices('(),;=<>!', k=random.randint(1, 20)))
        else:
            # Mutate valid statement
            stmt = self.generate_valid_statement()
            # Remove random parts or add junk
            if random.random() < 0.5 and len(stmt) > 10:
                cut_pos = random.randint(0, len(stmt) - 5)
                return stmt[:cut_pos]
            else:
                return stmt + random.choice([' !!!', ' @#$', ' SELECT', ';;'])


def fuzz(parse_fn, grammar_text=None, target_name=None):
    \"\"\"
    Generate test inputs and feed them to parse_fn.
    
    Args:
        parse_fn: The parser function to fuzz (call it with a string)
        grammar_text: Optional BNF grammar text for the target
        target_name: Name of the target being fuzzed (e.g. \"sql\", \"toml\")
    \"\"\"
    if target_name == \"sql\":
        fuzzer = SQLFuzzer(target_name)
        
        # Generate diverse inputs
        for i in range(1000):
            try:
                # Mostly valid inputs, some invalid
                if i < 800:
                    # Valid statements - add semicolon sometimes
                    stmt = fuzzer.generate_valid_statement()
                    if random.random() < 0.7:
                        stmt += ';'
                    parse_fn(stmt)
                else:
                    # Invalid inputs for error handling
                    parse_fn(fuzzer.generate_invalid_input())
            except Exception:
                pass
    else:
        # Fallback to random fuzzing for unknown targets
        for _ in range(1000):
            length = random.randint(1, 100)
            input_str = ''.join(random.choices(string.printable, k=length))
            try:
                parse_fn(input_str)
            except Exception:
                pass
"
}
